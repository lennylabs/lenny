type resp struct {
	status int
	filter string
	body   string
	err    string
}

func connect(auth string) (net.Conn, *bufio.Reader, error) {
	c, err := net.DialTimeout("tcp", proxy, 5*time.Second)
	if err != nil {
		return nil, nil, err
	}
	c.SetDeadline(time.Now().Add(20 * time.Second))
	fmt.Fprintf(c, "CONNECT %s HTTP/1.1\r\nHost: %s\r\n\r\n", auth, auth)
	br := bufio.NewReader(c)
	r, err := http.ReadResponse(br, &http.Request{Method: "CONNECT"})
	if err != nil {
		c.Close()
		return nil, nil, err
	}
	if r.StatusCode != 200 {
		c.Close()
		return nil, nil, fmt.Errorf("CONNECT status %d", r.StatusCode)
	}
	return c, br, nil
}

func tlsTo(auth string, proto string, insecure bool) (*tls.Conn, error) {
	c, _, err := connect(auth)
	if err != nil {
		return nil, err
	}
	host := auth
	if h, _, e := net.SplitHostPort(auth); e == nil {
		host = h
	}
	tc := tls.Client(c, &tls.Config{RootCAs: roots, ServerName: strings.ToLower(strings.TrimSuffix(host, ".")), NextProtos: []string{proto}, InsecureSkipVerify: insecure})
	if err := tc.Handshake(); err != nil {
		c.Close()
		return nil, err
	}
	return tc, nil
}

func readH1(br *bufio.Reader, method string) resp {
	r, err := http.ReadResponse(br, &http.Request{Method: method})
	if err != nil {
		return resp{err: "no response: " + err.Error()}
	}
	b, _ := io.ReadAll(io.LimitReader(r.Body, 4096))
	r.Body.Close()
	return resp{status: r.StatusCode, filter: r.Header.Get("X-Lenny-Filter"), body: string(b)}
}

// h2 one-connection client over raw frames, so that odd pseudo-headers go out as written.
type h2c struct {
	c    net.Conn
	f    *http2.Framer
	enc  *hpack.Encoder
	buf  bytes.Buffer
	next uint32
	dec  *hpack.Decoder
}

func newH2(auth string, insecure bool) (*h2c, error) {
	tc, err := tlsTo(auth, "h2", insecure)
	if err != nil {
		return nil, err
	}
	if tc.ConnectionState().NegotiatedProtocol != "h2" {
		tc.Close()
		return nil, fmt.Errorf("h2 not negotiated")
	}
	io.WriteString(tc, http2.ClientPreface)
	h := &h2c{c: tc, f: http2.NewFramer(tc, tc), next: 1}
	h.f.WriteSettings()
	h.enc = hpack.NewEncoder(&h.buf)
	h.dec = hpack.NewDecoder(65536, nil)
	return h, nil
}

func (h *h2c) do(hs [][2]string, body []byte, trailers ...[2]string) resp {
	id := h.next
	h.next += 2
	h.buf.Reset()
	for _, x := range hs {
		h.enc.WriteField(hpack.HeaderField{Name: x[0], Value: x[1]})
	}
	h.f.WriteHeaders(http2.HeadersFrameParam{StreamID: id, BlockFragment: h.buf.Bytes(), EndHeaders: true, EndStream: len(body) == 0})
	for off := 0; off < len(body); {
		n := len(body) - off
		if n > 16000 {
			n = 16000
		}
		end := off+n == len(body) && len(trailers) == 0
		if err := h.f.WriteData(id, end, body[off:off+n]); err != nil {
			return resp{err: "write: " + err.Error()}
		}
		off += n
	}
	if len(trailers) > 0 {
		h.buf.Reset()
		for _, x := range trailers {
			h.enc.WriteField(hpack.HeaderField{Name: x[0], Value: x[1]})
		}
		h.f.WriteHeaders(http2.HeadersFrameParam{StreamID: id, BlockFragment: h.buf.Bytes(), EndHeaders: true, EndStream: true})
	}
	var out resp
	var b bytes.Buffer
	for {
		fr, err := h.f.ReadFrame()
		if err != nil {
			out.err = "read: " + err.Error()
			return out
		}
		switch x := fr.(type) {
		case *http2.SettingsFrame:
			if !x.IsAck() {
				h.f.WriteSettingsAck()
			}
		case *http2.PingFrame:
			if !x.IsAck() {
				h.f.WritePing(true, x.Data)
			}
		case *http2.GoAwayFrame:
			out.err = "GOAWAY " + x.ErrCode.String()
			return out
		case *http2.RSTStreamFrame:
			if x.StreamID == id {
				out.err = "RST_STREAM " + x.ErrCode.String()
				return out
			}
		case *http2.WindowUpdateFrame:
		case *http2.HeadersFrame:
			fs, _ := h.dec.DecodeFull(x.HeaderBlockFragment())
			if x.StreamID != id {
				continue
			}
			for _, f := range fs {
				if f.Name == ":status" {
					fmt.Sscan(f.Value, &out.status)
				}
				if f.Name == "x-lenny-filter" {
					out.filter = f.Value
				}
			}
			if x.StreamEnded() {
				out.body = b.String()
				return out
			}
		case *http2.DataFrame:
			if x.StreamID == id {
				b.Write(x.Data())
				if len(x.Data()) > 0 {
					h.f.WriteWindowUpdate(0, uint32(len(x.Data())))
					h.f.WriteWindowUpdate(id, uint32(len(x.Data())))
				}
				if x.StreamEnded() {
					out.body = b.String()
					return out
				}
			}
		}
	}
}

type rec struct {
	Method  string              `json:"method"`
	URI     string              `json:"uri"`
	Host    string              `json:"host"`
	Proto   string              `json:"proto"`
	Headers map[string][]string `json:"headers"`
	BodyLen int                 `json:"bodyLen"`
	GQL     []struct {
		Ops []struct {
			Type   string   `json:"type"`
			Fields []string `json:"fields"`
		} `json:"ops"`
		OpsB []struct {
			Type   string   `json:"type"`
			Fields []string `json:"fields"`
		} `json:"opsB"`
	} `json:"gql"`
	Status int `json:"status"`
}

func drain() []rec {
	r, err := (&http.Client{Timeout: 5 * time.Second, Transport: &http.Transport{Proxy: nil}}).Get(stubAdmin + "/drain")
	if err != nil {
		fmt.Fprintln(os.Stderr, "drain:", err)
		return nil
	}
	defer r.Body.Close()
	var out []rec
	dec := json.NewDecoder(r.Body)
	for dec.More() {
		var x rec
		if dec.Decode(&x) != nil {
			break
		}
		out = append(out, x)
	}
	return out
}

type step struct {
	Method, Target string
	Hdr            []string
	Body           []byte
	Host           string // Host or :authority, default the CONNECT authority
	Raw            []byte // HTTP/1.1 only, written as is
	H2             [][2]string
	Trailers       [][2]string
}
type kase struct {
	ID, Group, Expect string // refuse, allow, nocred
	Conn              string
	Proto             string // both, h1, h2, plain
	Insecure          bool
	Steps             []step
	StepExpect        []string
	Note              string
}

var host, restP, gqlP string

func gj(q string, extra ...any) []byte {
	m := map[string]any{"query": q}
	for i := 0; i+1 < len(extra); i += 2 {
		m[extra[i].(string)] = extra[i+1]
	}
	var b bytes.Buffer
	e := json.NewEncoder(&b)
	e.SetEscapeHTML(false)
	e.Encode(m)
	return b.Bytes()
}

func gql(body []byte, hdr ...string) step {
	return step{Method: "POST", Target: gqlP, Body: body, Hdr: append([]string{"Content-Type: application/json"}, hdr...)}
}
func rest(m, p string, body []byte, hdr ...string) step {
	s := step{Method: m, Target: restP + p, Body: body, Hdr: hdr}
	if body != nil {
		s.Hdr = append([]string{"Content-Type: application/json"}, hdr...)
	}
	return s
}
func one(id, group, expect string, s step, note ...string) kase {
	n := ""
	if len(note) > 0 {
		n = note[0]
	}
	return kase{ID: id, Group: group, Expect: expect, Proto: "both", Steps: []step{s}, Note: n}
}
func h1only(id, group, expect string, raw string, note ...string) kase {
	n := ""
	if len(note) > 0 {
		n = note[0]
	}
	return kase{ID: id, Group: group, Expect: expect, Proto: "h1", Steps: []step{{Raw: []byte(raw)}}, Note: n}
}
func h2only(id, group, expect string, hs [][2]string, body []byte, note ...string) kase {
	n := ""
	if len(note) > 0 {
		n = note[0]
	}
	return kase{ID: id, Group: group, Expect: expect, Proto: "h2", Steps: []step{{H2: hs, Body: body}}, Note: n}
}

func gz(b []byte) []byte {
	var o bytes.Buffer
	w := gzip.NewWriter(&o)
	w.Write(b)
	w.Close()
	return o.Bytes()
}
func zs(b []byte) []byte {
	var o bytes.Buffer
	w, _ := zstd.NewWriter(&o)
	w.Write(b)
	w.Close()
	return o.Bytes()
}
func df(b []byte) []byte {
	var o bytes.Buffer
	w, _ := flate.NewWriter(&o, 5)
	w.Write(b)
	w.Close()
	return o.Bytes()
}

func rawChunked(target string, body []byte) []byte {
	var b bytes.Buffer
	fmt.Fprintf(&b, "POST %s HTTP/1.1\r\nHost: %s\r\nContent-Type: application/json\r\nTransfer-Encoding: chunked\r\n\r\n", target, host)
	for off := 0; off < len(body); off += 8192 {
		e := off + 8192
		if e > len(body) {
			e = len(body)
		}
		fmt.Fprintf(&b, "%x\r\n%s\r\n", e-off, body[off:e])
	}
	b.WriteString("0\r\n\r\n")
	return b.Bytes()
}

func rawChunkedSplit(target string, body []byte, at string) []byte {
	i := bytes.Index(body, []byte(at)) + len(at)
	var b bytes.Buffer
	fmt.Fprintf(&b, "POST %s HTTP/1.1\r\nHost: %s\r\nContent-Type: application/json\r\nTransfer-Encoding: chunked\r\n\r\n", target, host)
	fmt.Fprintf(&b, "%x\r\n%s\r\n%x\r\n%s\r\n0\r\n\r\n", i, body[:i], len(body)-i, body[i:])
	return b.Bytes()
}

func h2base(m, p, au string) [][2]string {
	return [][2]string{{":method", m}, {":scheme", "https"}, {":path", p}, {":authority", au}, {"content-type", "application/json"}, {"authorization", "token " + placeholder}}
}

func stepH1(s step, connHost string) []byte {
	if s.Raw != nil {
		return s.Raw
	}
	h := s.Host
	if h == "" {
		h = connHost
	}
	var b bytes.Buffer
	fmt.Fprintf(&b, "%s %s HTTP/1.1\r\nHost: %s\r\nAuthorization: token %s\r\nUser-Agent: attack\r\nAccept: */*\r\n", s.Method, s.Target, h, placeholder)
	for _, x := range s.Hdr {
		b.WriteString(x + "\r\n")
	}
	if s.Body != nil {
		fmt.Fprintf(&b, "Content-Length: %d\r\n", len(s.Body))
	}
	b.WriteString("\r\n")
	b.Write(s.Body)
	return b.Bytes()
}

func stepH2(s step, connHost string) ([][2]string, []byte) {
	if s.H2 != nil {
		return s.H2, s.Body
	}
	h := s.Host
	if h == "" {
		h = connHost
	}
	out := [][2]string{{":method", s.Method}, {":scheme", "https"}, {":path", s.Target}, {":authority", h}, {"authorization", "token " + placeholder}, {"user-agent", "attack"}, {"accept", "*/*"}}
	for _, x := range s.Hdr {
		k, v, _ := strings.Cut(x, ":")
		out = append(out, [2]string{strings.ToLower(strings.TrimSpace(k)), strings.TrimSpace(v)})
	}
	if s.Body != nil {
		out = append(out, [2]string{"content-length", fmt.Sprint(len(s.Body))})
	}
	return out, s.Body
}

type result struct {
	ID, Group, Proto, Expect, Verdict string
	Step                              int
	Status                            int
	Filter                            string
	Reason                            string
	Err                               string
	StubN                             int
	RealCred                          bool
	Placeholder                       bool
	StubURI                           string
	StubFields                        []string
	Note                              string
}

var realTok = map[string]string{"ghes.example.test": "GH-REAL-55aa01", "api.github.com": "GHCOM-REAL-77cc02", "gitlab.example.test": "GL-REAL-66dd04", "api.example.test": "ACME-REAL-9f2c71"}

func judge(k kase, proto string, step int, expect string, r resp, recs []rec) result {
	o := result{ID: k.ID, Group: k.Group, Proto: proto, Expect: expect, Step: step, Status: r.status, Filter: r.filter, Err: r.err, StubN: len(recs), Note: k.Note}
	var m struct {
		Message string `json:"message"`
	}
	if json.Unmarshal([]byte(r.body), &m) == nil && m.Message != "" {
		o.Reason = m.Message
	} else if r.status >= 400 {
		o.Reason = strings.TrimSpace(r.body)
	}
	for _, x := range recs {
		o.StubURI = x.Method + " " + x.URI
		for _, v := range x.Headers {
			for _, y := range v {
				for _, t := range realTok {
					if strings.Contains(y, t) {
						o.RealCred = true
					}
				}
				if strings.Contains(y, placeholder) {
					o.Placeholder = true
				}
			}
		}
		for _, g := range x.GQL {
			for _, op := range g.Ops {
				o.StubFields = append(o.StubFields, op.Type+":"+strings.Join(op.Fields, ","))
			}
		}
	}
	overrideSeen := false
	for _, x := range recs {
		for k := range x.Headers {
			if strings.EqualFold(k, "X-Http-Method-Override") {
				overrideSeen = true
			}
		}
	}
	if strings.Contains(r.err, "connection refused") || strings.Contains(r.err, "dial tcp") {
		o.Verdict = "INVALID-netagent-down"
		return o
	}
	switch expect {
	case "allow":
		if overrideSeen {
			o.Verdict = "FAIL-override-header-reached-stub"
		} else if r.status >= 200 && r.status < 300 && len(recs) == 1 && o.RealCred && !o.Placeholder {
			o.Verdict = "PASS"
		} else {
			o.Verdict = "FAIL-not-allowed"
		}
	case "refuse":
		if len(recs) > 0 && o.RealCred {
			o.Verdict = "FAIL-BYPASS"
		} else if len(recs) > 0 {
			o.Verdict = "FAIL-reached-no-cred"
		} else if r.status >= 200 && r.status < 300 {
			o.Verdict = "FAIL-2xx-no-stub"
		} else {
			o.Verdict = "PASS"
		}
	case "nocred":
		if o.RealCred {
			o.Verdict = "FAIL-BYPASS"
		} else if len(recs) > 0 {
			o.Verdict = "PASS-reached-no-cred"
		} else {
			o.Verdict = "PASS-did-not-reach"
		}
	}
	return o
}

func run(k kase, proto string) []result {
	var out []result
	conn := k.Conn
	if conn == "" {
		conn = host + ":443"
	}
	connHost := conn
	if h, p, e := net.SplitHostPort(conn); e == nil && p == "443" {
		connHost = h
	}
	exp := func(i int) string {
		if k.StepExpect != nil {
			return k.StepExpect[i]
		}
		return k.Expect
	}
	settle := func() []rec {
		time.Sleep(15 * time.Millisecond)
		return drain()
	}
	drain()
	switch proto {
	case "h1":
		tc, err := tlsTo(conn, "http/1.1", k.Insecure)
		if err != nil {
			return []result{judge(k, proto, 0, exp(0), resp{err: err.Error()}, settle())}
		}
		defer tc.Close()
		tc.SetDeadline(time.Now().Add(20 * time.Second))
		br := bufio.NewReader(tc)
		for i, s := range k.Steps {
			tc.Write(stepH1(s, connHost))
			m := s.Method
			if m == "" {
				m = "POST"
				if bytes.HasPrefix(s.Raw, []byte("OPTIONS")) || bytes.HasPrefix(s.Raw, []byte("GET")) {
					m = "GET"
				}
			}
			r := readH1(br, m)
			out = append(out, judge(k, proto, i, exp(i), r, settle()))
		}
	case "h2":
		h, err := newH2(conn, k.Insecure)
		if err != nil {
			return []result{judge(k, proto, 0, exp(0), resp{err: err.Error()}, settle())}
		}
		defer h.c.Close()
		for i, s := range k.Steps {
			hs, b := stepH2(s, connHost)
			r := h.do(hs, b, s.Trailers...)
			out = append(out, judge(k, proto, i, exp(i), r, settle()))
		}
	case "plain":
		c, err := net.DialTimeout("tcp", proxy, 5*time.Second)
		if err != nil {
			return []result{judge(k, proto, 0, exp(0), resp{err: err.Error()}, settle())}
		}
		defer c.Close()
		c.SetDeadline(time.Now().Add(20 * time.Second))
		c.Write(k.Steps[0].Raw)
		r := readH1(bufio.NewReader(c), "POST")
		out = append(out, judge(k, proto, 0, exp(0), r, settle()))
	case "pipe", "framed":
		tc, err := tlsTo(conn, "http/1.1", k.Insecure)
		if err != nil {
			return []result{judge(k, proto, 0, exp(0), resp{err: err.Error()}, settle())}
		}
		defer tc.Close()
		tc.SetDeadline(time.Now().Add(20 * time.Second))
		var w bytes.Buffer
		for _, s := range k.Steps {
			w.Write(stepH1(s, connHost))
		}
		tc.Write(w.Bytes())
		br := bufio.NewReader(tc)
		var rs []resp
		for range k.StepExpect {
			rs = append(rs, readH1(br, "POST"))
		}
		recs := settle()
		// Two responses, one stub total: the allowed request must be the only one that arrived.
		for i, r := range rs {
			var sub []rec
			if exp(i) == "allow" {
				sub = recs
			} else if len(recs) > 1 {
				sub = recs[1:]
			}
			out = append(out, judge(k, proto, i, exp(i), r, sub))
		}
	}
	return out
}

func trunc(s string, n int) string {
	s = strings.TrimSpace(strings.ReplaceAll(s, "\n", " "))
	if len(s) > n {
		return s[:n]
	}
	return s
}

// latency alternates the same request between an unfiltered bound host and the filtered one on persistent connections.
func latency(class string, n, rate int) {
	body := map[string]func() []byte{
		"create": func() []byte { return gj(goodM, "variables", goodVars) },
	}[class]
	var b []byte
	if body != nil {
		b = body()
	} else {
		d, err := os.ReadFile(class)
		if err != nil {
			panic(err)
		}
		b = d
	}
	a, err := newH2("api.example.test:443", false)
	if err != nil {
		panic(err)
	}
	f, err := newH2(host+":443", false)
	if err != nil {
		panic(err)
	}
	var ta, tf []time.Duration
	tick := time.NewTicker(time.Second / time.Duration(rate))
	defer tick.Stop()
	once := func(h *h2c, au string) time.Duration {
		hs := append(h2base("POST", gqlP, au), [2]string{"content-length", fmt.Sprint(len(b))})
		t := time.Now()
		r := h.do(hs, b)
		d := time.Since(t)
		if r.status != 200 {
			fmt.Println("status", r.status, r.err, trunc(r.body, 200))
		}
		return d
	}
	for i := 0; i < n; i++ {
		<-tick.C
		if i%2 == 0 {
			ta = append(ta, once(a, "api.example.test"))
			tf = append(tf, once(f, host))
		} else {
			tf = append(tf, once(f, host))
			ta = append(ta, once(a, "api.example.test"))
		}
		if i%50 == 49 {
			drain()
		}
	}
	drain()
	pc := func(x []time.Duration, p float64) float64 {
		s := append([]time.Duration{}, x...)
		sort.Slice(s, func(i, j int) bool { return s[i] < s[j] })
		return float64(s[int(float64(len(s)-1)*p)].Microseconds()) / 1000
	}
	var diffs []time.Duration
	for i := range ta {
		diffs = append(diffs, tf[i]-ta[i])
	}
	fmt.Printf("body=%d bytes pairs=%d rate=%d/s\n unfiltered p50=%.2fms p90=%.2fms p99=%.2fms\n filtered   p50=%.2fms p90=%.2fms p99=%.2fms\n paired difference p50=%.3fms p90=%.3fms\n", len(b), n, rate, pc(ta, .5), pc(ta, .9), pc(ta, .99), pc(tf, .5), pc(tf, .9), pc(tf, .99), pc(diffs, .5), pc(diffs, .9))
}
