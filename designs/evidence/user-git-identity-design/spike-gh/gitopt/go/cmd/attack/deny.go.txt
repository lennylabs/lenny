var denyMode bool

type oracleRest struct {
	Method, Path string
	re           *regexp.Regexp
}

var oracleM = map[string]bool{}
var oracleR []oracleRest

// loadOracle reads the preset's names. The check below is a separate reading: it decodes and folds the path the stub saw and searches it, rather than calling the filter's matcher.
func loadOracle(f, fl string) {
	b, err := os.ReadFile(f)
	if err != nil {
		panic(err)
	}
	var m map[string]struct {
		DenyMutations []string `json:"denyMutations"`
		DenyRest      []struct{ Method, Path string } `json:"denyRest"`
	}
	json.Unmarshal(b, &m)
	k := "ghes"
	if fl == "ghcom" || fl == "ghe" {
		k = "ghcom"
	}
	for _, x := range m[k].DenyMutations {
		oracleM[x] = true
	}
	for _, r := range m[k].DenyRest {
		p := strings.TrimPrefix(r.Path, ".../")
		p = strings.TrimSuffix(p, "/**")
		q := regexp.QuoteMeta(p)
		q = regexp.MustCompile(`\\\{[a-z]+\\\}`).ReplaceAllString(q, `[^/]+`)
		end := "$"
		if strings.HasSuffix(r.Path, "/**") {
			end = "(/.*)?$"
		}
		oracleR = append(oracleR, oracleRest{r.Method, r.Path, regexp.MustCompile("(^|/)" + q + end)})
	}
}

func oracleNorm(u string) string {
	p, _, _ := strings.Cut(u, "?")
	for i := 0; i < 3; i++ {
		if d, err := url.PathUnescape(p); err == nil {
			p = d
		}
	}
	p = strings.ToLower(strings.ReplaceAll(p, "\\", "/"))
	p = regexp.MustCompile(`;[^/]*`).ReplaceAllString(p, "")
	p = regexp.MustCompile(`/+`).ReplaceAllString(p, "/")
	for strings.Contains(p, "/./") {
		p = strings.ReplaceAll(p, "/./", "/")
	}
	for {
		n := regexp.MustCompile(`/[^/]+/\.\./`).ReplaceAllString(p, "/")
		if n == p {
			break
		}
		p = n
	}
	return strings.TrimRight(p, "/")
}

func deniedAt(x rec) string {
	for _, g := range x.GQL {
		for _, o := range g.Ops {
			for _, f := range o.Fields {
				if oracleM[f] {
					return "gql:" + f
				}
			}
		}
		for _, o := range g.OpsB {
			for _, f := range o.Fields {
				if oracleM[f] {
					return "gql:" + f
				}
			}
		}
	}
	ms := map[string]bool{strings.ToUpper(x.Method): true}
	for k, vs := range x.Headers {
		if strings.EqualFold(k, "X-Http-Method-Override") || strings.EqualFold(k, "X-Http-Method") || strings.EqualFold(k, "X-Method-Override") {
			for _, v := range vs {
				ms[strings.ToUpper(v)] = true
			}
		}
	}
	u, _ := url.Parse(x.URI)
	if u != nil {
		for _, v := range u.Query()["_method"] {
			ms[strings.ToUpper(v)] = true
		}
	}
	if v, err := url.ParseQuery(x.Body); err == nil {
		for _, m := range v["_method"] {
			ms[strings.ToUpper(m)] = true
		}
	}
	n := oracleNorm(x.URI)
	for _, r := range oracleR {
		if ms[strings.ToUpper(r.Method)] && r.re.MatchString(n) {
			return "rest:" + r.Method + " " + r.Path
		}
	}
	return ""
}

func judgeDeny(o result, k kase, expect string, r resp, recs []rec) result {
	// keep the fields judge already filled by re-reading the stub records
	var m struct {
		Message string `json:"message"`
	}
	if json.Unmarshal([]byte(r.body), &m) == nil && m.Message != "" {
		o.Reason = m.Message
	}
	for _, x := range recs {
		o.StubURI = x.Method + " " + x.URI
		for _, v := range x.Headers {
			for _, y := range v {
				for _, t := range realTok {
					if strings.Contains(y, t) {
						o.RealCred = true
					}
				}
				if strings.Contains(y, placeholder) {
					o.Placeholder = true
				}
			}
		}
		for _, g := range x.GQL {
			for _, op := range g.Ops {
				o.StubFields = append(o.StubFields, op.Type+":"+strings.Join(op.Fields, ","))
			}
		}
	}
	o.StubN = len(recs)
	act := ""
	for _, x := range recs {
		if d := deniedAt(x); d != "" {
			act = d
		}
	}
	switch {
	case act != "" && o.RealCred:
		o.Verdict = "REACHED-DENIED-ACTION"
		o.Note = act
	case act != "":
		o.Verdict = "REACHED-denied-action-no-credential"
		o.Note = act
	case len(recs) > 0 && o.RealCred:
		o.Verdict = "REACHED-not-denied"
	case len(recs) > 0:
		o.Verdict = "REACHED-no-credential"
	case expect == "allow" || k.Group == "deny-allow":
		o.Verdict = "PASS-no-stub-expected-allow"
	case o.Err != "" && strings.Contains(o.Err, "dial tcp"):
		o.Verdict = "INVALID-netagent-down"
	default:
		o.Verdict = "PASS-refused"
	}
	return o
}

func hexb(b byte) string {
	const h = "0123456789abcdef"
	return string([]byte{h[b>>4], h[b&15]})
}
