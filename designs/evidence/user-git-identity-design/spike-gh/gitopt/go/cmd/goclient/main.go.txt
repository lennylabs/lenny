// goclient: go-git clone and push, with its defaults, against the URL it is given.
package main

import (
	"fmt"
	"os"

	git "github.com/go-git/go-git/v5"
)

func main() {
	url, dir := os.Args[1], os.Args[2]
	r, err := git.PlainClone(dir, false, &git.CloneOptions{URL: url, Depth: 1})
	if err != nil {
		fmt.Println("clone error:", err)
		os.Exit(1)
	}
	h, _ := r.Head()
	fmt.Println("cloned", h.Hash().String()[:12])
}
