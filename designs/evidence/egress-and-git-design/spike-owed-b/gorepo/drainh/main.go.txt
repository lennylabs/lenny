// drainh: the repository's drain stamp and consumer, driven against a kind API server with the real CRDs, under impersonated identities.
// It calls podclaim.StampDrainRequest as the gateway and warmpool.PodReconciler.Reconcile as the controller.
package main

import (
	"context"
	"fmt"
	"os"
	"time"

	corev1 "k8s.io/api/core/v1"
	metav1 "k8s.io/apimachinery/pkg/apis/meta/v1"
	"k8s.io/apimachinery/pkg/runtime"
	clientgoscheme "k8s.io/client-go/kubernetes/scheme"
	"k8s.io/client-go/rest"
	"k8s.io/client-go/tools/clientcmd"
	ctrl "sigs.k8s.io/controller-runtime"
	"sigs.k8s.io/controller-runtime/pkg/client"

	"github.com/lennylabs/lenny/pkg/admission/ownership"
	lennyv1 "github.com/lennylabs/lenny/pkg/apis/lenny/v1alpha1"
	"github.com/lennylabs/lenny/pkg/controller/warmpool"
	"github.com/lennylabs/lenny/pkg/gateway/podlifecycle/podclaim"
	"github.com/lennylabs/lenny/pkg/sandbox/state"
)

const ns = "agents"

func cl(s *runtime.Scheme, cfg *rest.Config, as string) client.Client {
	c := rest.CopyConfig(cfg)
	if as != "" {
		c.Impersonate = rest.ImpersonationConfig{UserName: as}
	}
	x, err := client.New(c, client.Options{Scheme: s})
	if err != nil {
		panic(err)
	}
	return x
}

func main() {
	cfg, err := clientcmd.BuildConfigFromFlags("", os.Args[1])
	if err != nil {
		panic(err)
	}
	s := runtime.NewScheme()
	_ = clientgoscheme.AddToScheme(s)
	_ = lennyv1.AddToScheme(s)
	admin := cl(s, cfg, "")
	gw := cl(s, cfg, "system:serviceaccount:lenny-system:lenny-gateway")
	wpc := cl(s, cfg, "system:serviceaccount:lenny-system:lenny-controller")
	ctx := context.Background()
	fmt.Printf("state.ValidTransitions reserved->draining allowed: %v\n", hasEdge(state.Reserved, state.Draining))
	fmt.Printf("state.ValidTransitions claimed->draining allowed: %v\n", hasEdge(state.Claimed, state.Draining))
	fmt.Printf("state.ValidTransitions idle->draining allowed: %v\n", hasEdge(state.Idle, state.Draining))

	for _, phase := range []state.State{state.Idle, state.Reserved, state.Claimed} {
		name := "drain-" + string(phase)
		// The pod carries group-a from an earlier assignment, and the tenant's current assignment is group-b: a mismatch.
		p := &corev1.Pod{ObjectMeta: metav1.ObjectMeta{Name: name, Namespace: ns, Labels: map[string]string{
			"lenny.dev/managed": "true", "lenny.dev/egress-group": "group-a"}},
			Spec: corev1.PodSpec{Containers: []corev1.Container{{Name: "c", Image: "ob-tools:1", ImagePullPolicy: corev1.PullNever, Command: []string{"ob", "sleep"}}}}}
		if err := wpc.Create(ctx, p); err != nil {
			fmt.Printf("create pod %s as controller: %v\n", name, err)
			continue
		}
		sb := &lennyv1.Sandbox{ObjectMeta: metav1.ObjectMeta{Name: name, Namespace: ns}}
		sb.Spec.PoolRef = "pool"
		sb.Spec.RuntimeRef = "rt"
		if err := admin.Create(ctx, sb); err != nil {
			fmt.Printf("create sandbox: %v\n", err)
			continue
		}
		// The controller is the sole writer of Sandbox.status, so the fixture sets the phase under its field manager.
		st := &lennyv1.Sandbox{TypeMeta: metav1.TypeMeta{APIVersion: lennyv1.GroupVersion.String(), Kind: "Sandbox"}, ObjectMeta: metav1.ObjectMeta{Name: name, Namespace: ns}}
		st.Status.Phase = string(phase)
		if err := wpc.Status().Patch(ctx, st, client.Apply, client.FieldOwner(string(ownership.WarmPoolController))); err != nil {
			fmt.Printf("set sandbox phase: %v\n", err)
		}
		if phase != state.Idle {
			fmt.Printf("[%s] the pod holds a claim in the design's rebind and routing paths\n", phase)
		}
		// Refuse at the claim point, as the design says, and stamp through the repository's stamp function as the gateway.
		assign := "group-b"
		if p.Labels["lenny.dev/egress-group"] != assign {
			err := podclaim.StampDrainRequest(ctx, gw, ns, name, time.Now())
			fmt.Printf("[%s] mismatch refused; StampDrainRequest as gateway: err=%v\n", phase, err)
		}
		r := &warmpool.PodReconciler{Client: wpc}
		res, err := r.Reconcile(ctx, ctrl.Request{NamespacedName: client.ObjectKey{Namespace: ns, Name: name}})
		var after lennyv1.Sandbox
		_ = admin.Get(ctx, client.ObjectKey{Namespace: ns, Name: name}, &after)
		fmt.Printf("[%s] real PodReconciler.Reconcile as controller: res=%+v err=%v; Sandbox phase is now %q\n", phase, res, err, after.Status.Phase)
	}
}

func hasEdge(a, b state.State) bool {
	for _, t := range state.ValidTransitions() {
		if t.From == a && t.To == b {
			return true
		}
	}
	return false
}
