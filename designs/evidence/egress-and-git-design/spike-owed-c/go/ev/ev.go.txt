// Rec is the common envelope from the design's "Evidence streams".
type Rec struct {
	Stream        string          `json:"stream"`
	SchemaVersion int             `json:"schemaVersion"`
	Producer      string          `json:"producer"`
	Incarnation   string          `json:"incarnation"`
	Sequence      uint64          `json:"sequence"`
	SequenceEnd   uint64          `json:"sequenceEnd,omitempty"` // a chunk, aggregate, or gap: the last sequence it covers
	Kind          string          `json:"kind,omitempty"`    // "", "gap", or "denial_aggregate"
	ObservedAt    time.Time       `json:"observedAt"`
	TenantId      string          `json:"tenantId"`
	PoolId        string          `json:"poolId,omitempty"`
	PodUid        string          `json:"podUid,omitempty"`
	SessionId     string          `json:"sessionId,omitempty"`
	Environment   string          `json:"environment,omitempty"`
	TraceId       string          `json:"traceId,omitempty"`
	Provenance    string          `json:"provenance"`
	PayloadKeyRef string          `json:"payloadKeyRef,omitempty"`
	Payload       json.RawMessage `json:"payload"`
}

func Hash(s string) uint32 {
	h := fnv.New32a()
	h.Write([]byte(s))
	return h.Sum32()
}

// Replica hashes a tenant to an ingest replica.
func Replica(tenant string, n int) int { return int(Hash(tenant) % uint32(n)) }

// UUIDFrom returns a stable UUID-formatted value for a seed.
func UUIDFrom(seed string) string {
	h1, h2 := fnv.New64a(), fnv.New64()
	h1.Write([]byte(seed))
	h2.Write([]byte(seed))
	var b [16]byte
	copy(b[:8], h1.Sum(nil))
	copy(b[8:], h2.Sum(nil))
	x := hex.EncodeToString(b[:])
	return x[:8] + "-" + x[8:12] + "-" + x[12:16] + "-" + x[16:20] + "-" + x[20:]
}

func RandHex(r *rand.Rand, n int) string {
	const c = "0123456789abcdef"
	b := make([]byte, n)
	for i := range b {
		b[i] = c[r.IntN(16)]
	}
	return string(b)
}

var vendors = []string{"amazonaws.com", "cloudfront.net", "githubusercontent.com", "googleapis.com", "azure.com", "blob.core.windows.net", "openai.azure.com", "anthropic.com", "npmjs.org", "pypi.org", "pythonhosted.org", "docker.io", "gcr.io", "cloudflare.com", "fastly.net", "akamaized.net", "okta.com", "datadoghq.com", "sentry.io", "slack.com"}
var words = []string{"api", "prod", "eu-west-1", "us-east-1", "ap-southeast-2", "cdn", "assets", "storage", "registry", "objects", "uploads", "edge", "internal", "static", "svc", "data", "ml", "inference", "ingest", "telemetry"}

// Host returns an authority drawn from a long-tailed population: most are short vendor names, some carry a
// hashed label as CDNs and object stores do, and about one in a hundred approaches the length bound.
func Host(r *rand.Rand) string {
	v := vendors[r.IntN(len(vendors))]
	switch p := r.IntN(100); {
	case p < 45:
		return words[r.IntN(len(words))] + "." + v
	case p < 80:
		return fmt.Sprintf("%s.%s.%s", words[r.IntN(len(words))], words[r.IntN(len(words))], v)
	case p < 99:
		return fmt.Sprintf("%s-%s.%s.%s", RandHex(r, 12), words[r.IntN(len(words))], words[r.IntN(len(words))], v)
	default:
		var sb strings.Builder
		for sb.Len() < 200 {
			sb.WriteString(RandHex(r, 30))
			sb.WriteByte('.')
		}
		sb.WriteString(v)
		return sb.String()
	}
}

var rules = []string{"tenant-policy/engineering-default/allow-registries", "platform/baseline/deny-metadata", "tenant-policy/ml-team/allow-model-apis", "pool-overlay/warm-default/allow-package-mirrors", "tenant-policy/security/credential-pinned/llm", "platform/shadow-2026-q4/observe"}

// Session gives a tenant a stable small set of sessions, pods, pools, and traces.
func Session(tenant string, n int) (session, pod, pool, trace string) {
	s := fmt.Sprintf("%s/%d", tenant, n)
	session, pod, pool = UUIDFrom("s"+s), UUIDFrom("p"+s), "pool-"+UUIDFrom("o"+tenant)[:18]
	x := UUIDFrom("t" + s)
	trace = strings.ReplaceAll(x, "-", "")
	return
}

// DecisionPayload follows the design's egress.decision field list.
func DecisionPayload(r *rand.Rand, did, authority, peer, outcome string) json.RawMessage {
	shadow := ""
	if r.IntN(10) == 0 {
		shadow = "deny"
	}
	b, _ := json.Marshal(map[string]any{"decisionId": did, "authority": authority, "port": 443, "peerAddress": peer, "outcome": outcome,
		"rule": rules[r.IntN(len(rules))], "routeTarget": "direct", "egressGroup": "group-" + RandHex(r, 6), "shadowOutcome": shadow})
	return b
}

// TunnelPayload follows the design's egress.tunnel field list.
func TunnelPayload(did, upAddr string, srcAddr string, srcPort uint32, start, end time.Time, up, down uint64, how string) json.RawMessage {
	b, _ := json.Marshal(map[string]any{"decisionId": did, "upstreamAddress": upAddr, "sourceAddress": srcAddr, "upstreamSourcePort": srcPort, "start": start.UTC().Format(time.RFC3339Nano), "end": end.UTC().Format(time.RFC3339Nano), "bytesUp": up, "bytesDown": down, "endedBy": how})
	return b
}

// ClientTLS builds an mTLS client config from a leaf directory and CA file.
func ClientTLS(dir, ca, serverName string) *tls.Config {
	c, err := tls.LoadX509KeyPair(dir+"/tls.crt", dir+"/tls.key")
	if err != nil {
		panic(err)
	}
	b, err := os.ReadFile(ca)
	if err != nil {
		panic(err)
	}
	p := x509.NewCertPool()
	p.AppendCertsFromPEM(b)
	return &tls.Config{Certificates: []tls.Certificate{c}, RootCAs: p, ServerName: serverName, NextProtos: []string{"h2"}}
}

// PGPool connects through the pooler in transaction mode: no prepared statements survive a checkout, so it uses
// pgx's parameterized exec mode and no session state.
func PGPool(ctx context.Context, url string, maxConns int32) *pgxpool.Pool {
	cfg, err := pgxpool.ParseConfig(url)
	if err != nil {
		panic(err)
	}
	cfg.MaxConns = maxConns
	cfg.ConnConfig.DefaultQueryExecMode = pgx.QueryExecModeExec
	p, err := pgxpool.NewWithConfig(ctx, cfg)
	if err != nil {
		panic(err)
	}
	return p
}

// Hour is the object prefix and manifest key for a time.
func Hour(t time.Time) string { return t.UTC().Format("2006010215") }
