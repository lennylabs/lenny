// memwatch: reads a container's cgroup from outside that container, so polling adds nothing to the limited cgroup and an
// out-of-memory kill cannot take the poller with it. Every reading is appended to the log as soon as it is taken, and a
// failed poll adds a line rather than replacing an earlier one.
//   time_ms current peak anon sock kernel file oom_kill oom goroutines heapInuse stackInuse active terminated opaque
package main

import (
	"encoding/json"
	"flag"
	"fmt"
	"net/http"
	"os"
	"strconv"
	"strings"
	"time"
)

func num(path string) int64 {
	b, err := os.ReadFile(path)
	if err != nil {
		return -1
	}
	n, _ := strconv.ParseInt(strings.TrimSpace(string(b)), 10, 64)
	return n
}
func kv(path string) map[string]int64 {
	m := map[string]int64{}
	b, err := os.ReadFile(path)
	if err != nil {
		return m
	}
	for _, l := range strings.Split(string(b), "\n") {
		f := strings.Fields(l)
		if len(f) == 2 {
			m[f[0]], _ = strconv.ParseInt(f[1], 10, 64)
		}
	}
	return m
}
func main() {
	cg := flag.String("cg", "", "")
	logf := flag.String("log", "", "")
	stats := flag.String("stats", "http://127.0.0.1:9100/stats", "")
	every := flag.Duration("every", 250*time.Millisecond, "")
	flag.Parse()
	f, err := os.OpenFile(*logf, os.O_CREATE|os.O_APPEND|os.O_WRONLY, 0o644)
	if err != nil {
		panic(err)
	}
	c := &http.Client{Timeout: 200 * time.Millisecond}
	for range time.Tick(*every) {
		st, ev := kv(*cg+"/memory.stat"), kv(*cg+"/memory.events")
		var s struct {
			Goroutines, HeapInuse, StackInuse, Active, Terminated, Opaque int64
		}
		s.Goroutines = -1
		if r, err := c.Get(*stats); err == nil {
			json.NewDecoder(r.Body).Decode(&s)
			r.Body.Close()
		}
		fmt.Fprintf(f, "%d %d %d %d %d %d %d %d %d %d %d %d %d %d %d\n", time.Now().UnixMilli(), num(*cg+"/memory.current"), num(*cg+"/memory.peak"), st["anon"], st["sock"], st["kernel"], st["file"], ev["oom_kill"], ev["oom"], s.Goroutines, s.HeapInuse, s.StackInuse, s.Active, s.Terminated, s.Opaque)
	}
}
