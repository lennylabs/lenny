// evidence: lenny-evidence as a separate service, per the design's "Evidence streams".
//   POST /v1/records   asynchronous ingest, NDJSON envelopes
//   POST /v1/commit    a regulated tenant's decision, acknowledged after a group commit to the write-ahead table
// Producers authenticate with mTLS, and the verified SAN selects the streams they may write. Ingest overwrites `producer`.
// Tenants hash to replicas. Per-tenant buffers, denials aggregated per host, a batch flushes at a size bound or at FLUSH.

func envDur(k string, d time.Duration) time.Duration {
	if x, err := time.ParseDuration(os.Getenv(k)); err == nil {
		return x
	}
	return d
}
func envStr(k, d string) string {
	if x := os.Getenv(k); x != "" {
		return x
	}
	return d
}
func envInt(k string, d int) int {
	if x, err := strconv.Atoi(os.Getenv(k)); err == nil {
		return x
	}
	return d
}

type key struct{ tenant, stream, producer, incarnation, session string }

// aggr covers a contiguous sequence range of repeated denials of one host. The first denial is kept whole, and the aggregate holds the repeats.
type aggr struct {
	host         string
	rec          ev.Rec
	firstDid     string
	count        int
	seqFirst     uint64
	seqLast      uint64
	first1, last time.Time
}

type bat struct {
	key
	recs      [][]byte
	size      int
	opened    time.Time
	seqF      uint64
	seqL      uint64
	tF, tL    time.Time
	open      *aggr
	walIDs    []int64
	chunks    map[string]*chunk
	nRec      int
	hourLabel string
}

// chunk gathers one session's payloads in a batch. They compress before encryption.
type chunk struct {
	first, last uint64
	tF          time.Time
	rec         ev.Rec
	buf         bytes.Buffer
}

type tenant struct {
	mu       sync.Mutex
	name     string
	bats     map[key]*bat
	bytes    int64
	inflight bool
	drops    map[key]*[2]uint64 // first and last dropped sequence, to become a gap record
	queued   []*bat
}

const shards = 64

var (
	tmu  [shards]sync.RWMutex
	tmap [shards]map[string]*tenant
)

func getT(name string) *tenant {
	s := ev.Hash(name) % shards
	tmu[s].RLock()
	t := tmap[s][name]
	tmu[s].RUnlock()
	if t != nil {
		return t
	}
	tmu[s].Lock()
	defer tmu[s].Unlock()
	if t = tmap[s][name]; t == nil {
		t = &tenant{name: name, bats: map[key]*bat{}, drops: map[key]*[2]uint64{}}
		tmap[s][name] = t
	}
	return t
}

func eachT(f func(*tenant)) {
	for s := 0; s < shards; s++ {
		tmu[s].RLock()
		ts := make([]*tenant, 0, len(tmap[s]))
		for _, t := range tmap[s] {
			ts = append(ts, t)
		}
		tmu[s].RUnlock()
		for _, t := range ts {
			f(t)
		}
	}
}

// ---- session keys. A destroyed key leaves the cache at once on this replica and through a push on the others.
// A cached copy also expires at keyTTL, which bounds a lost push. A later record for that session gets a new key.
type cachedKey struct {
	ref string
	k   []byte
	at  time.Time
}

var (
	keyMu  sync.Mutex
	keyMap = map[string]*cachedKey{}
)

func sessionKey(ctx context.Context, tenantID, session string) (string, []byte, error) {
	id := tenantID + "/" + session
	keyMu.Lock()
	defer keyMu.Unlock()
	if c := keyMap[id]; c != nil && time.Since(c.at) < keyTTL {
		return c.ref, c.k, nil
	}
	tx, err := pg.Begin(ctx)
	if err != nil {
		return "", nil, err
	}
	defer tx.Rollback(ctx)
	if _, err := tx.Exec(ctx, `select set_config('app.current_tenant',$1,true)`, tenantID); err != nil {
		return "", nil, err
	}
	var ref string
	var out []byte
	err = tx.QueryRow(ctx, `select key_ref, key from evidence_session_keys where tenant_id=$1 and session_id=$2 and destroyed_at is null`, tenantID, session).Scan(&ref, &out)
	if err == pgx.ErrNoRows {
		var gen int
		if err := tx.QueryRow(ctx, `select coalesce(max(gen),0)+1 from evidence_session_keys where tenant_id=$1 and session_id=$2`, tenantID, session).Scan(&gen); err != nil {
			return "", nil, err
		}
		k := make([]byte, 32)
		rand.Read(k)
		ref = fmt.Sprintf("k/%s/%s/%d", tenantID, session, gen)
		err = tx.QueryRow(ctx, `insert into evidence_session_keys(key_ref,tenant_id,session_id,gen,key) values($1,$2,$3,$4,$5) on conflict(tenant_id,session_id) where destroyed_at is null do update set key_ref=evidence_session_keys.key_ref returning key_ref, key`, ref, tenantID, session, gen, k).Scan(&ref, &out)
		if err != nil {
			return "", nil, err
		}
		nKeyNew.Add(1)
	} else if err != nil {
		return "", nil, err
	}
	if err := tx.Commit(ctx); err != nil {
		return "", nil, err
	}
	keyMap[id] = &cachedKey{ref, out, time.Now()}
	return ref, out, nil
}

func invalidateKey(tenantID, session string) {
	keyMu.Lock()
	delete(keyMap, tenantID+"/"+session)
	keyMu.Unlock()
}

// purgeSession drops a session's plaintext still in this replica's buffers, and leaves a gap record in its place.
func purgeSession(tenantID, session string) int {
	t := getT(tenantID)
	t.mu.Lock()
	defer t.mu.Unlock()
	n := 0
	purge := func(b *bat) {
		c := b.chunks[session]
		if c == nil {
			return
		}
		n += bytes.Count(c.buf.Bytes(), []byte{'\n'})
		delete(b.chunks, session)
		nPurged.Add(1)
		d := &[2]uint64{c.first, c.last}
		pl, _ := json.Marshal(map[string]any{"from": d[0], "to": d[1], "reason": "erased_before_flush"})
		r := ev.Rec{Stream: b.stream, SchemaVersion: 1, Producer: b.producer, Incarnation: b.incarnation, Sequence: d[0], SequenceEnd: d[1], Kind: "gap", ObservedAt: time.Now().UTC(), TenantId: t.name, SessionId: session, Provenance: "enforced", Payload: pl}
		line, _ := json.Marshal(&r)
		b.recs = append(b.recs, line)
	}
	for _, b := range t.bats {
		purge(b)
	}
	for _, b := range t.queued {
		purge(b)
	}
	return n
}

func encrypt(k, plain []byte) []byte {
	b, _ := aes.NewCipher(k)
	g, _ := cipher.NewGCM(b)
	n := make([]byte, g.NonceSize())
	rand.Read(n)
	return g.Seal(n, n, plain, nil)
}

// ---- ingest
func (t *tenant) add(r *ev.Rec, line []byte, walID int64) {
	sess := ""
	if perSession && encStreams[r.Stream] {
		sess = r.SessionId
	}
	k := key{t.name, r.Stream, r.Producer, r.Incarnation, sess}
	t.mu.Lock()
	defer t.mu.Unlock()
	nRecords.Add(1)
	if walID == 0 && r.Kind != "gap" && (t.bytes >= tenantBound || buffered.Load() >= globalBound) {
		if buffered.Load() >= globalBound {
			nDroppedG.Add(1)
		} else {
			nDroppedT.Add(1)
		}
		d := t.drops[k]
		if d == nil {
			d = &[2]uint64{r.Sequence, r.Sequence}
			t.drops[k] = d
		}
		d[1] = r.Sequence
		return
	}
	b := t.bats[k]
	if b == nil {
		b = &bat{key: k, opened: time.Now(), seqF: r.Sequence, tF: r.ObservedAt, hourLabel: ev.Hour(r.ObservedAt)}
		t.bats[k] = b
	}
	if h := ev.Hour(r.ObservedAt); h != b.hourLabel && (len(b.recs) > 0 || b.open != nil) {
		// a batch stays within one hour of observation
		t.detach(b)
		b = &bat{key: k, opened: time.Now(), seqF: r.Sequence, tF: r.ObservedAt, hourLabel: h}
		t.bats[k] = b
	}
	if r.Sequence < b.seqF {
		b.seqF = r.Sequence
	}
	if r.Sequence > b.seqL {
		b.seqL = r.Sequence
	}
	if b.tF.After(r.ObservedAt) {
		b.tF = r.ObservedAt
	}
	if r.ObservedAt.After(b.tL) {
		b.tL = r.ObservedAt
	}
	if walID != 0 {
		b.walIDs = append(b.walIDs, walID)
	}
	if chunkMode && encStreams[r.Stream] {
		if b.chunks == nil {
			b.chunks = map[string]*chunk{}
		}
		c := b.chunks[r.SessionId]
		if c == nil {
			c = &chunk{first: r.Sequence, tF: r.ObservedAt, rec: *r}
			b.chunks[r.SessionId] = c
		}
		c.last = r.Sequence
		c.buf.Write(r.Payload)
		c.buf.WriteByte('\n')
		b.size += len(r.Payload) + 1
		t.bytes += int64(len(r.Payload) + 1)
		buffered.Add(int64(len(r.Payload) + 1))
		return
	}
	// Denials aggregate over a contiguous range of one host. Any other record in the position closes the aggregate, so a loss between denials stays visible.
	if aggregate && r.Stream == "egress.decision" && walID == 0 && r.Kind == "" {
		var p struct {
			Outcome    string `json:"outcome"`
			Authority  string `json:"authority"`
			DecisionId string `json:"decisionId"`
		}
		json.Unmarshal(r.Payload, &p)
		if p.Outcome == "deny" {
			if a := b.open; a != nil && a.host == p.Authority && a.seqLast+1 == r.Sequence {
				a.count++
				a.seqLast = r.Sequence
				a.last = r.ObservedAt
				nAggIn.Add(1)
				return
			}
			b.closeAgg()
			b.open = &aggr{host: p.Authority, rec: *r, firstDid: p.DecisionId, count: 0, seqFirst: r.Sequence, seqLast: r.Sequence, first1: r.ObservedAt, last: r.ObservedAt}
		} else {
			b.closeAgg()
		}
	} else if b.open != nil {
		b.closeAgg()
	}
	c := append([]byte(nil), line...)
	b.recs = append(b.recs, c)
	b.size += len(c)
	t.bytes += int64(len(c))
	buffered.Add(int64(len(c)))
	if b.size >= sizeBound {
		t.detach(b)
	}
}

// closeAgg writes the open aggregate's repeats as one record with a sequence range. The first denial is already in the batch.
func (b *bat) closeAgg() {
	a := b.open
	if a == nil {
		return
	}
	b.open = nil
	if a.seqLast == a.seqFirst {
		return
	}
	pl, _ := json.Marshal(map[string]any{"authority": a.host, "count": a.seqLast - a.seqFirst, "firstDecisionId": a.firstDid, "firstAt": a.first1, "lastAt": a.last})
	r := a.rec
	r.Kind, r.Sequence, r.SequenceEnd, r.ObservedAt, r.Payload = "denial_aggregate", a.seqFirst+1, a.seqLast, a.last, pl
	line, _ := json.Marshal(&r)
	b.recs = append(b.recs, line)
	b.size += len(line)
	nAggOut.Add(1)
}

func (t *tenant) detach(b *bat) {
	b.closeAgg()
	delete(t.bats, b.key)
	t.queued = append(t.queued, b)
}

// ingest decodes each line, overwrites producer from the verified SAN, and applies the per-session key when the stream has one.
func prep(ctx context.Context, san string, line []byte) (*ev.Rec, []byte, error) {
	var r ev.Rec
	if err := json.Unmarshal(line, &r); err != nil {
		return nil, nil, err
	}
	r.Producer = san
	if encStreams[r.Stream] && !chunkMode {
		ref, k, err := sessionKey(ctx, r.TenantId, r.SessionId)
		if err != nil {
			return nil, nil, err
		}
		ct := encrypt(k, r.Payload)
		nEncWrites.Add(1)
		r.PayloadKeyRef = ref
		r.Payload, _ = json.Marshal(ct)
	}
	out, err := json.Marshal(&r)
	return &r, out, err
}

func sanOf(req *http.Request) string {
	if req.TLS == nil || len(req.TLS.PeerCertificates) == 0 || len(req.TLS.PeerCertificates[0].DNSNames) == 0 {
		return ""
	}
	return req.TLS.PeerCertificates[0].DNSNames[0]
}

func handleRecords(w http.ResponseWriter, req *http.Request) {
	if draining.Load() {
		http.Error(w, "draining", 503)
		return
	}
	san := sanOf(req)
	streams := allow[san]
	if streams == nil {
		nForbidden.Add(1)
		http.Error(w, "client not allowed", 403)
		return
	}
	body, err := io.ReadAll(req.Body)
	if err != nil {
		http.Error(w, "read", 400)
		return
	}
	nBytesIn.Add(int64(len(body)))
	var refused int
	for len(body) > 0 {
		i := bytes.IndexByte(body, '\n')
		var line []byte
		if i < 0 {
			line, body = body, nil
		} else {
			line, body = body[:i], body[i+1:]
		}
		if len(line) == 0 {
			continue
		}
		r, out, err := prep(req.Context(), san, line)
		if err != nil {
			refused++
			continue
		}
		if !streams[r.Stream] {
			refused++
			nForbidden.Add(1)
			continue
		}
		if san == "lenny-git-mirror.lenny-system.svc" && !mirrorRoutes[r.TenantId] {
			refused++
			nForbidden.Add(1)
			continue
		}
		getT(r.TenantId).add(r, out, 0)
	}
	if refused > 0 {
		w.WriteHeader(403)
		fmt.Fprintf(w, "refused %d records\n", refused)
		return
	}
	fmt.Fprintln(w, "ok")
}

var mirrorRoutes = map[string]bool{}

// ---- flush workers. One batch per tenant in flight, so a busy tenant cannot occupy every worker.
var jobs = make(chan *job, 4096)

type job struct {
	t *tenant
	b *bat
}

func flusher() {
	for range time.Tick(250 * time.Millisecond) {
		now := time.Now()
		eachT(func(t *tenant) {
			t.mu.Lock()
			for _, b := range t.bats {
				if now.Sub(b.opened) >= flushEvery {
					t.detach(b)
				}
			}
			for k, d := range t.drops {
				gapRecord(t, k, d)
				delete(t.drops, k)
			}
			var j *job
			if !t.inflight && len(t.queued) > 0 {
				j = &job{t, t.queued[0]}
				t.queued = t.queued[1:]
				t.inflight = true
			}
			t.mu.Unlock()
			if j != nil {
				jobs <- j
			}
		})
	}
}

// gapRecord writes what ingest dropped past the tenant's bound. Caller holds t.mu.
func gapRecord(t *tenant, k key, d *[2]uint64) {
	nGap.Add(1)
	pl, _ := json.Marshal(map[string]any{"from": d[0], "to": d[1], "reason": "ingest_tenant_bound"})
	r := ev.Rec{Stream: k.stream, SchemaVersion: 1, Producer: k.producer, Incarnation: k.incarnation, Sequence: d[0], SequenceEnd: d[1], Kind: "gap", ObservedAt: time.Now().UTC(), TenantId: t.name, SessionId: k.session, Provenance: "enforced", Payload: pl}
	line, _ := json.Marshal(&r)
	b := t.bats[k]
	if b == nil {
		b = &bat{key: k, opened: time.Now(), seqF: d[0], tF: r.ObservedAt, hourLabel: ev.Hour(r.ObservedAt)}
		t.bats[k] = b
	}
	if d[0] < b.seqF {
		b.seqF = d[0]
	}
	if d[1] > b.seqL {
		b.seqL = d[1]
	}
	b.tL = r.ObservedAt
	b.closeAgg()
	b.recs = append(b.recs, line)
	b.size += len(line)
	t.bytes += int64(len(line))
	buffered.Add(int64(len(line)))
}

func worker(ctx context.Context) {
	var z bytes.Buffer
	for j := range jobs {
		flushBatch(ctx, j, &z)
		j.t.mu.Lock()
		j.t.inflight = false
		j.t.mu.Unlock()
	}
}

type manRow struct {
	session, batchID, tenant, stream, hour, producer, inc, key, digest string
	seqF, seqL                                                int64
	tF, tL                                                    time.Time
	bytes, raw                                                int64
	recs                                                      int
	walIDs                                                    []int64
	done                                                      chan error
}

var manCh = make(chan *manRow, 8192)

func flushBatch(ctx context.Context, j *job, z *bytes.Buffer) {
	b := j.b
	z.Reset()
	g, _ := gzip.NewWriterLevel(z, gzip.DefaultCompression)
	raw := 0
	nrec := 0
	for _, r := range b.recs {
		g.Write(r)
		g.Write([]byte{'\n'})
		raw += len(r) + 1
		nrec++
	}
	// chunks: compress, then encrypt under the session's key
	sess := make([]string, 0, len(b.chunks))
	for k := range b.chunks {
		sess = append(sess, k)
	}
	sort.Strings(sess)
	for _, k := range sess {
		c := b.chunks[k]
		ref, key, err := sessionKey(ctx, b.tenant, k)
		if err != nil {
			log.Printf("chunk key: %v", err)
			continue
		}
		var cz bytes.Buffer
		cg := gzip.NewWriter(&cz)
		cg.Write(c.buf.Bytes())
		cg.Close()
		if dropChunk(b.tenant, k) { // a test hook: this chunk is lost in place, with nothing recorded
			continue
		}
		r := c.rec
		r.Producer, r.Sequence, r.SequenceEnd, r.PayloadKeyRef = b.producer, c.first, c.last, ref
		nEncWrites.Add(1)
		r.Payload, _ = json.Marshal(encrypt(key, cz.Bytes()))
		line, _ := json.Marshal(&r)
		g.Write(line)
		g.Write([]byte{'\n'})
		raw += len(line) + 1
		nrec++
	}
	g.Close()
	d := sha256.Sum256(z.Bytes())
	dg := hex.EncodeToString(d[:])
	id := fmt.Sprintf("%s-%d-%s", writerInc, b.seqF, dg[:16])
	sessLabel := b.session
	objKey := fmt.Sprintf("%s/evidence/%s/%s/%s.gz", b.tenant, b.stream, b.hourLabel, id)
	for {
		c, cancel := context.WithTimeout(ctx, 10*time.Second)
		_, err := mc.PutObject(c, bucket, objKey, bytes.NewReader(z.Bytes()), int64(z.Len()), minio.PutObjectOptions{ContentType: "application/gzip", DisableMultipart: true, DisableContentSha256: true})
		cancel()
		if err == nil {
			break
		}
		nPutFail.Add(1)
		time.Sleep(500 * time.Millisecond)
	}
	m := &manRow{session: sessLabel, batchID: id, tenant: b.tenant, stream: b.stream, hour: b.hourLabel, producer: b.producer, inc: b.incarnation, key: objKey, digest: dg, seqF: int64(b.seqF), seqL: int64(b.seqL), tF: b.tF, tL: b.tL, bytes: int64(z.Len()), raw: int64(raw), recs: nrec, walIDs: b.walIDs, done: make(chan error, 1)}
	for {
		manCh <- m
		err := <-m.done
		if err == nil {
			break
		}
		log.Printf("manifest: %v", err)
		time.Sleep(500 * time.Millisecond)
		m.done = make(chan error, 1)
	}
	nBatches.Add(1)
	nBytesOut.Add(int64(z.Len()))
	nRawOut.Add(int64(raw))
	j.t.mu.Lock()
	j.t.bytes -= int64(b.size)
	j.t.mu.Unlock()
	buffered.Add(-int64(b.size))
	publishDurable(b)
}

// manifestWriter puts batches' rows in one transaction, with the write-ahead rows they drained.
func manifestWriter(ctx context.Context) {
	for {
		first := <-manCh
		rows := []*manRow{first}
		t := time.After(50 * time.Millisecond)
	fill:
		for len(rows) < 500 {
			select {
			case m := <-manCh:
				rows = append(rows, m)
			case <-t:
				break fill
			}
		}
		err := writeManifest(ctx, rows)
		for _, m := range rows {
			m.done <- err
		}
	}
}

func writeManifest(ctx context.Context, rows []*manRow) error {
	if !manifestPT {
		return writeManifestTx(ctx, rows, "__all__")
	}
	by := map[string][]*manRow{}
	for _, m := range rows {
		by[m.tenant] = append(by[m.tenant], m)
	}
	var wg sync.WaitGroup
	var firstErr atomic.Value
	sem := make(chan struct{}, 8)
	for t, g := range by {
		wg.Add(1)
		sem <- struct{}{}
		go func(t string, g []*manRow) {
			defer wg.Done()
			defer func() { <-sem }()
			if err := writeManifestTx(ctx, g, t); err != nil {
				firstErr.Store(err)
			}
		}(t, g)
	}
	wg.Wait()
	nManifestTx.Add(int64(len(by)))
	if e := firstErr.Load(); e != nil {
		return e.(error)
	}
	return nil
}

var nManifestTx atomic.Int64

func writeManifestTx(ctx context.Context, rows []*manRow, tc string) error {
	n := len(rows)
	a := make([][]string, 9)
	for i := range a {
		a[i] = make([]string, n)
	}
	seqF, seqL, by, raw := make([]int64, n), make([]int64, n), make([]int64, n), make([]int64, n)
	recs := make([]int32, n)
	tF, tL := make([]time.Time, n), make([]time.Time, n)
	var wal []int64
	for i, m := range rows {
		a[0][i], a[1][i], a[2][i], a[3][i], a[4][i], a[5][i], a[6][i], a[7][i], a[8][i] = m.batchID, m.tenant, m.stream, m.hour, m.producer, m.inc, m.key, m.digest, m.session
		seqF[i], seqL[i], by[i], raw[i], recs[i], tF[i], tL[i] = m.seqF, m.seqL, m.bytes, m.raw, int32(m.recs), m.tF, m.tL
		wal = append(wal, m.walIDs...)
	}
	c, cancel := context.WithTimeout(ctx, 15*time.Second)
	defer cancel()
	tx, err := pg.Begin(c)
	if err != nil {
		return err
	}
	defer tx.Rollback(c)
	if _, err := tx.Exec(c, `select set_config('app.current_tenant',$1,true)`, tc); err != nil {
		return err
	}
	_, err = tx.Exec(c, `insert into manifest(batch_id,tenant_id,stream,hour,producer,incarnation,object_key,digest,seq_first,seq_last,bytes,raw_bytes,records,time_first,time_last,writer,session)
	 select b,t,s,h,p,i,k,d,sf,sl,by,r,n,tf,tl,$16,x.sess from unnest($1::text[],$2::text[],$3::text[],$4::text[],$5::text[],$6::text[],$7::text[],$8::text[],$9::bigint[],$10::bigint[],$11::bigint[],$12::bigint[],$13::int[],$14::timestamptz[],$15::timestamptz[],$17::text[]) as x(b,t,s,h,p,i,k,d,sf,sl,by,r,n,tf,tl,sess)`,
		a[0], a[1], a[2], a[3], a[4], a[5], a[6], a[7], seqF, seqL, by, raw, recs, tF, tL, writerInc, a[8])
	if err != nil {
		return err
	}
	if len(wal) > 0 {
		if _, err := tx.Exec(c, `delete from evidence_wal where id = any($1::bigint[])`, wal); err != nil {
			return err
		}
	}
	return tx.Commit(c)
}

// ---- durable sequence reports. After each flush ingest tells the producer each tenant and stream's durable sequence.
type durSub struct {
	mu      sync.Mutex
	pending map[string]uint64
	notify  chan struct{}
}

var (
	dsMu   sync.Mutex
	dsubs  = map[string]map[*durSub]struct{}{}
	closeC = make(chan struct{})
)

func publishDurable(b *bat) {
	dsMu.Lock()
	subs := dsubs[b.incarnation]
	dsMu.Unlock()
	if len(subs) == 0 {
		return
	}
	k := b.tenant + " " + b.stream
	dsMu.Lock()
	defer dsMu.Unlock()
	for s := range dsubs[b.incarnation] {
		s.mu.Lock()
		if b.seqL > s.pending[k] {
			s.pending[k] = b.seqL
		}
		s.mu.Unlock()
		select {
		case s.notify <- struct{}{}:
		default:
		}
	}
}

func handleDurable(w http.ResponseWriter, req *http.Request) {
	if allow[sanOf(req)] == nil {
		http.Error(w, "client not allowed", 403)
		return
	}
	inc := req.URL.Query().Get("incarnation")
	s := &durSub{pending: map[string]uint64{}, notify: make(chan struct{}, 1)}
	dsMu.Lock()
	if dsubs[inc] == nil {
		dsubs[inc] = map[*durSub]struct{}{}
	}
	dsubs[inc][s] = struct{}{}
	dsMu.Unlock()
	defer func() {
		dsMu.Lock()
		delete(dsubs[inc], s)
		dsMu.Unlock()
	}()
	fl := w.(http.Flusher)
	w.WriteHeader(200)
	fmt.Fprintf(w, "{\"writer\":%q}\n", writerInc)
	fl.Flush()
	send := func() bool {
		s.mu.Lock()
		p := s.pending
		s.pending = map[string]uint64{}
		s.mu.Unlock()
		if len(p) == 0 {
			return true
		}
		if err := json.NewEncoder(w).Encode(map[string]any{"durable": p}); err != nil {
			return false
		}
		fl.Flush()
		return true
	}
	tick := time.NewTicker(time.Second)
	defer tick.Stop()
	for {
		select {
		case <-s.notify:
			if !send() {
				return
			}
		case <-tick.C:
			if !send() {
				return
			}
		case <-closeC:
			send()
			fmt.Fprintln(w, `{"closing":true}`)
			fl.Flush()
			return
		case <-req.Context().Done():
			return
		}
	}
}

// a test hook for the per-session check: lose the next chunk of a session.
var (
	dropMu sync.Mutex
	dropN  = map[string]int{}
)

func dropChunk(tenantID, session string) bool {
	dropMu.Lock()
	defer dropMu.Unlock()
	k := tenantID + "/" + session
	if dropN[k] > 0 {
		dropN[k]--
		return true
	}
	return false
}

// ---- the regulated path: group commit to the write-ahead table through the pooler
type walItem struct {
	tenant string
	rec    string
	done   chan error
}

var walCh = make(chan walItem, 100000)

var (
	latMu                sync.Mutex
	groupSizes, groupDur []int64
)

func walLoop(ctx context.Context) {
	for {
		first := <-walCh
		batch := []walItem{first}
	fill:
		for len(batch) < walMax {
			select {
			case it := <-walCh:
				batch = append(batch, it)
			default:
				break fill
			}
		}
		t0 := time.Now()
		var err error
		var errs []error
		if walMode == "pertenant" {
			errs = commitPerTenant(ctx, batch)
		} else {
			err = commitAll(ctx, batch)
		}
		d := time.Since(t0)
		latMu.Lock()
		groupSizes = append(groupSizes, int64(len(batch)))
		groupDur = append(groupDur, d.Microseconds())
		latMu.Unlock()
		for i, it := range batch {
			e := err
			if errs != nil {
				e = errs[i]
			}
			if e != nil {
				nWalFail.Add(1)
			} else {
				nWalOK.Add(1)
			}
			it.done <- e
		}
	}
}

func commitAll(ctx context.Context, batch []walItem) error {
	ten, rec := make([]string, len(batch)), make([]string, len(batch))
	for i, it := range batch {
		ten[i], rec[i] = it.tenant, it.rec
	}
	c, cancel := context.WithTimeout(ctx, 2*time.Second)
	defer cancel()
	// BEGIN, the sentinel, the insert, and COMMIT go as one pipeline, so a group costs one round trip to the pooler.
	var b pgx.Batch
	b.Queue(`begin`)
	b.Queue(`select set_config('app.current_tenant','__all__',true)`)
	b.Queue(`insert into evidence_wal(replica,tenant_id,rec) select $1::int,t,r from unnest($2::text[],$3::text[]) as x(t,r)`, replica, ten, rec)
	b.Queue(`commit`)
	br := pg.SendBatch(c, &b)
	for i := 0; i < 4; i++ {
		if _, err := br.Exec(); err != nil {
			br.Close()
			return err
		}
	}
	return br.Close()
}

// commitPerTenant: each tenant in the group commits in its own transaction under its own tenant context (the design, §12.3's guard).
// An error reaches only that tenant's waiters.
func commitPerTenant(ctx context.Context, batch []walItem) []error {
	by := map[string][]int{}
	for i, it := range batch {
		by[it.tenant] = append(by[it.tenant], i)
	}
	c, cancel := context.WithTimeout(ctx, 2*time.Second)
	defer cancel()
	errs := make([]error, len(batch))
	var wg sync.WaitGroup
	sem := make(chan struct{}, 16)
	for t, idx := range by {
		wg.Add(1)
		sem <- struct{}{}
		go func(t string, idx []int) {
			defer wg.Done()
			defer func() { <-sem }()
			recs := make([]string, len(idx))
			for j, i := range idx {
				recs[j] = batch[i].rec
			}
			var b pgx.Batch
			b.Queue(`begin`)
			b.Queue(`select set_config('app.current_tenant',$1,true)`, t)
			b.Queue(`insert into evidence_wal(replica,tenant_id,rec) select $1::int,$2,r from unnest($3::text[]) as x(r)`, replica, t, recs)
			b.Queue(`commit`)
			br := pg.SendBatch(c, &b)
			var err error
			for i := 0; i < 4; i++ {
				if _, e := br.Exec(); e != nil {
					err = e
					break
				}
			}
			br.Close()
			if err != nil {
				for _, i := range idx {
					errs[i] = err
				}
			}
		}(t, idx)
	}
	wg.Wait()
	nCommitTx.Add(int64(len(by)))
	return errs
}

var nCommitTx atomic.Int64

// noteRefused handles a decision the producer has already refused or will refuse: if its commit landed nothing else is recorded, and if not a gap record names the refusal.
func noteRefused(r *ev.Rec, reason string, err error) {
	if err == nil {
		nLateLanded.Add(1)
		return
	}
	nLateLost.Add(1)
	pl, _ := json.Marshal(map[string]any{"from": r.Sequence, "to": r.Sequence, "reason": reason})
	g := ev.Rec{Stream: r.Stream, SchemaVersion: 1, Producer: r.Producer, Incarnation: r.Incarnation, Sequence: r.Sequence, SequenceEnd: r.Sequence, Kind: "gap", ObservedAt: time.Now().UTC(), TenantId: r.TenantId, Provenance: "enforced", Payload: pl}
	line, _ := json.Marshal(&g)
	getT(r.TenantId).add(&g, line, 0)
}

func handleCommit(w http.ResponseWriter, req *http.Request) {
	if draining.Load() {
		w.Header().Set("X-Lenny-Draining", "1")
		http.Error(w, "draining", 503)
		return
	}
	san := sanOf(req)
	if !allow[san]["egress.decision"] {
		nForbidden.Add(1)
		http.Error(w, "client not allowed", 403)
		return
	}
	body, err := io.ReadAll(req.Body)
	if err != nil {
		http.Error(w, "read", 400)
		return
	}
	r, out, err := prep(req.Context(), san, body)
	if err != nil {
		http.Error(w, "bad record", 400)
		return
	}
	it := walItem{r.TenantId, string(out), make(chan error, 1)}
	select {
	case walCh <- it:
	default:
		noteRefused(r, "refused_queue_full", fmt.Errorf("queue full"))
		http.Error(w, "queue full", 503)
		return
	}
	select {
	case err := <-it.done:
		if err != nil {
			noteRefused(r, "refused_commit_failed", err)
			http.Error(w, "commit failed", 503)
			return
		}
		w.WriteHeader(200)
	case <-req.Context().Done():
		// The producer's bound passed. The commit may still land.
		go func() { noteRefused(r, "refused_commit_late", <-it.done) }()
	}
}

// drainLoop moves committed write-ahead rows into the normal batch path. Their rows are deleted in the same
// transaction as the manifest row, so a crash re-reads them rather than losing them.
func drainLoop(ctx context.Context) {
	var hw int64
	for range time.Tick(envDur("DRAIN_EVERY", 5*time.Second)) {
		for !draining.Load() {
			c, cancel := context.WithTimeout(ctx, 10*time.Second)
			tx, err := pg.Begin(c)
			if err != nil {
				cancel()
				break
			}
			tx.Exec(c, `select set_config('app.current_tenant','__all__',true)`)
			rows, err := tx.Query(c, `select id, rec from evidence_wal where replica=$1 and id>$2 order by id limit 20000`, replica, hw)
			n := 0
			if err == nil {
				for rows.Next() {
					var id int64
					var s string
					if rows.Scan(&id, &s) == nil {
						var r ev.Rec
						if json.Unmarshal([]byte(s), &r) == nil {
							getT(r.TenantId).add(&r, []byte(s), id)
							nDrained.Add(1)
						}
						hw = id
						n++
					}
				}
				rows.Close()
			}
			tx.Rollback(c)
			cancel()
			if n < 20000 {
				break
			}
		}
	}
}

// ---- anchors: each interval, a digest of the tenant's new manifest rows goes into the chain as one event.
var (
	chainMu                    sync.Mutex
	nReplicas                  = envInt("REPLICAS_N", 2)
	anchorEvents, anchorMicros atomic.Int64
)

func anchorLoop(ctx context.Context) {
	for range time.Tick(envDur("ANCHOR", 300*time.Second)) {
		if err := anchorOnce(ctx); err != nil {
			log.Printf("anchor: %v", err)
		}
	}
}

func anchorOnce(ctx context.Context) error {
	t0 := time.Now()
	c, cancel := context.WithTimeout(ctx, 60*time.Second)
	defer cancel()
	tx, err := pg.Begin(c)
	if err != nil {
		return err
	}
	defer tx.Rollback(c)
	if _, err := tx.Exec(c, `select set_config('app.current_tenant','__all__',true)`); err != nil {
		return err
	}
	rows, err := tx.Query(c, `select tenant_id, id, batch_id, digest from manifest where not anchored order by tenant_id, id limit 400000`)
	if err != nil {
		return err
	}
	type it struct{ id, dg string }
	by := map[string][]it{}
	var ids []int64
	for rows.Next() {
		var t, b, d string
		var id int64
		if err := rows.Scan(&t, &id, &b, &d); err != nil {
			rows.Close()
			return err
		}
		if ev.Replica(t, nReplicas) != replica {
			continue
		}
		by[t] = append(by[t], it{b, d})
		ids = append(ids, id)
	}
	rows.Close()
	if err := rows.Err(); err != nil {
		return err
	}
	if len(by) == 0 {
		return nil
	}
	chainMu.Lock()
	defer chainMu.Unlock()
	names := make([]string, 0, len(by))
	for t := range by {
		names = append(names, t)
	}
	type st struct {
		seq  int64
		hash string
	}
	head := map[string]st{}
	hr, err := tx.Query(c, `select distinct on (tenant_id) tenant_id, seq, hash from audit_chain where tenant_id = any($1::text[]) order by tenant_id, seq desc`, names)
	if err != nil {
		return err
	}
	for hr.Next() {
		var t string
		var x st
		if err := hr.Scan(&t, &x.seq, &x.hash); err != nil {
			hr.Close()
			return err
		}
		head[t] = x
	}
	hr.Close()
	var tn, dg, pv, hs, de []string
	var sq []int64
	for t, items := range by {
		h := sha256.New()
		var bs []string
		for _, x := range items {
			fmt.Fprintf(h, "%s %s\n", x.id, x.dg)
			bs = append(bs, x.id)
		}
		d := hex.EncodeToString(h.Sum(nil))
		cur := head[t]
		x := sha256.Sum256([]byte(cur.hash + "|manifest|" + d))
		hash := hex.EncodeToString(x[:])
		detail, _ := json.Marshal(map[string]any{"batches": bs})
		if os.Getenv("ANCHOR_BATCHES") == "0" {
			detail = []byte(`{"digestOnly":true}`)
		}
		tn, dg, pv, hs, de, sq = append(tn, t), append(dg, d), append(pv, cur.hash), append(hs, hash), append(de, string(detail)), append(sq, cur.seq+1)
	}
	if _, err := tx.Exec(c, `insert into audit_chain(tenant_id,seq,kind,digest,prev,hash,detail) select t,s,'manifest',d,p,h,x::jsonb from unnest($1::text[],$2::bigint[],$3::text[],$4::text[],$5::text[],$6::text[]) as y(t,s,d,p,h,x)`, tn, sq, dg, pv, hs, de); err != nil {
		return err
	}
	if _, err := tx.Exec(c, `update manifest set anchored=true where id = any($1::bigint[])`, ids); err != nil {
		return err
	}
	if err := tx.Commit(c); err != nil {
		return err
	}
	anchorEvents.Add(int64(len(tn)))
	anchorMicros.Add(time.Since(t0).Microseconds())
	return nil
}

// ---- egress.accesslog: Envoy calls lenny-evidence directly under its own identity. Envoy's stream cannot hash per tenant,
// so the replica that receives a record writes it, under a position that names that replica's incarnation.
type alsSrv struct {
	alsv3.UnimplementedAccessLogServiceServer
}

var (
	alsSeqMu sync.Mutex
	alsSeq   = map[string]uint64{}
	nAlsIn, nAlsForeign atomic.Int64
	alsStop = func() {}
)

func (alsSrv) StreamAccessLogs(st grpc.ClientStreamingServer[alsv3.StreamAccessLogsMessage, alsv3.StreamAccessLogsResponse]) error {
	san := ""
	if p, ok := peer.FromContext(st.Context()); ok {
		if ti, ok := p.AuthInfo.(grpccreds.TLSInfo); ok && len(ti.State.PeerCertificates) > 0 && len(ti.State.PeerCertificates[0].DNSNames) > 0 {
			san = ti.State.PeerCertificates[0].DNSNames[0]
		}
	}
	if !allow[san]["egress.accesslog"] {
		nForbidden.Add(1)
		return fmt.Errorf("client not allowed")
	}
	for {
		m, err := st.Recv()
		if err != nil {
			if err == io.EOF {
				return nil
			}
			return err
		}
		for _, e := range m.GetHttpLogs().GetLogEntry() {
			cp := e.GetCommonProperties()
			fm := cp.GetMetadata().GetFilterMetadata()["envoy.filters.http.ext_authz"].GetFields()
			tn := fm["tenant"].GetStringValue()
			if tn == "" {
				continue
			}
			nAlsIn.Add(1)
			if ev.Replica(tn, nReplicas) != replica {
				nAlsForeign.Add(1)
			}
			alsSeqMu.Lock()
			alsSeq[tn]++
			sq := alsSeq[tn]
			alsSeqMu.Unlock()
			up, src := cp.GetUpstreamRemoteAddress().GetSocketAddress(), cp.GetUpstreamLocalAddress().GetSocketAddress()
			start := cp.GetStartTime().AsTime()
			pl := ev.TunnelPayload(fm["decision_id"].GetStringValue(), up.GetAddress(), src.GetAddress(), src.GetPortValue(), start, start.Add(cp.GetTimeToLastDownstreamTxByte().AsDuration()), cp.GetUpstreamWireBytesSent(), cp.GetUpstreamWireBytesReceived(), cp.GetAccessLogType().String())
			r := &ev.Rec{Stream: "egress.accesslog", SchemaVersion: 1, Producer: san, Incarnation: writerInc, Sequence: sq, ObservedAt: time.Now().UTC(), TenantId: tn, Provenance: "enforced", Payload: pl}
			line, _ := json.Marshal(r)
			getT(tn).add(r, line, 0)
		}
	}
}

func pct(v []int64, p float64) int64 {
	if len(v) == 0 {
		return 0
	}
	x := append([]int64(nil), v...)
	sort.Slice(x, func(i, j int) bool { return x[i] < x[j] })
	return x[int(float64(len(x)-1)*p)]
}

func flushAll() {
	eachT(func(t *tenant) {
		t.mu.Lock()
		for _, b := range t.bats {
			t.detach(b)
		}
		for k, d := range t.drops {
			gapRecord(t, k, d)
			delete(t.drops, k)
		}
		for _, b := range t.bats {
			t.detach(b)
		}
		t.mu.Unlock()
	})
}

	ctx := context.Background()
	var err error
	mc, err = minio.New(os.Getenv("MINIO"), &minio.Options{Creds: miniocreds.NewStaticV4("minioadmin", "minioadmin", "")})
	if err != nil {
		log.Fatal(err)
	}
	pg = ev.PGPool(ctx, os.Getenv("PG"), int32(envInt("PGCONNS", 16)))
	go walLoop(ctx)
	if os.Getenv("DRAIN") != "off" {
		go drainLoop(ctx)
	}
	go manifestWriter(ctx)
	go flusher()
	go anchorLoop(ctx)
	for i := 0; i < workers; i++ {
		go worker(ctx)
	}
	mux := http.NewServeMux()
	mux.HandleFunc("/v1/records", handleRecords)
	mux.HandleFunc("/v1/commit", handleCommit)
	mux.HandleFunc("/v1/durable", handleDurable)
	mux.HandleFunc("/v1/read", handleRead)
	mux.HandleFunc("/v1/consumers/position", handlePosition)
	mux.HandleFunc("/healthz", func(w http.ResponseWriter, r *http.Request) {
		if draining.Load() {
			http.Error(w, "draining", 503)
			return
		}
		fmt.Fprintln(w, "ok")
	})
	caB, err := os.ReadFile(os.Getenv("CA"))
	if err != nil {
		log.Fatal(err)
	}
	p := x509.NewCertPool()
	p.AppendCertsFromPEM(caB)
	if _, err := pg.Exec(ctx, `insert into evidence_writers(writer, replica) values($1,$2)`, writerInc, replica); err != nil {
		log.Printf("writer row: %v", err)
	}
	crt, err := tls.LoadX509KeyPair(os.Getenv("LEAF")+"/tls.crt", os.Getenv("LEAF")+"/tls.key")
	if err != nil {
		log.Fatal(err)
	}
	go func() {
		l, err := net.Listen("tcp", ":8444")
		if err != nil {
			log.Fatal(err)
		}
		g := grpc.NewServer(grpc.Creds(grpccreds.NewTLS(&tls.Config{Certificates: []tls.Certificate{crt}, ClientAuth: tls.RequireAndVerifyClientCert, ClientCAs: p, MinVersion: tls.VersionTLS12, NextProtos: []string{"h2"}})))
		alsv3.RegisterAccessLogServiceServer(g, alsSrv{})
		alsStop = g.Stop
		log.Print(g.Serve(l))
	}()
	srv := &http.Server{Addr: ":8443", Handler: mux, TLSConfig: &tls.Config{ClientAuth: tls.RequireAndVerifyClientCert, ClientCAs: p, MinVersion: tls.VersionTLS12}, ErrorLog: log.New(io.Discard, "", 0)}
	go func() {
		m := http.NewServeMux()
		m.HandleFunc("/stats", stats)
		m.HandleFunc("/keys/destroy", func(w http.ResponseWriter, r *http.Request) {
			q := r.URL.Query()
			tn, se := q.Get("tenant"), q.Get("session")
			tx, err := pg.Begin(ctx)
			if err == nil {
				defer tx.Rollback(ctx)
				tx.Exec(ctx, `select set_config('app.current_tenant',$1,true)`, tn)
				_, err = tx.Exec(ctx, `update evidence_session_keys set key=null, destroyed_at=now() where tenant_id=$1 and session_id=$2 and destroyed_at is null`, tn, se)
				if err == nil {
					err = tx.Commit(ctx)
				}
			}
			invalidateKey(tn, se)
			if purgeOnDest {
				fmt.Fprintf(w, "purged %d buffered payloads; ", purgeSession(tn, se))
			}
			pushed := 0
			if pushInval {
				for _, p := range peers {
					if p == "" {
						continue
					}
					if resp, e := (&http.Client{Timeout: 2 * time.Second}).Get("http://" + p + ":9100/keys/invalidate?tenant=" + tn + "&session=" + se); e == nil {
						resp.Body.Close()
						pushed++
					}
				}
			}
			fmt.Fprintf(w, "destroyed err=%v pushed=%d\n", err, pushed)
		})
		m.HandleFunc("/keys/invalidate", func(w http.ResponseWriter, r *http.Request) {
			q := r.URL.Query()
			invalidateKey(q.Get("tenant"), q.Get("session"))
			if purgeOnDest {
				purgeSession(q.Get("tenant"), q.Get("session"))
			}
			fmt.Fprintln(w, "ok")
		})
		m.HandleFunc("/dropchunk", func(w http.ResponseWriter, r *http.Request) {
			dropMu.Lock()
			dropN[r.URL.Query().Get("tenant")+"/"+r.URL.Query().Get("session")]++
			dropMu.Unlock()
			fmt.Fprintln(w, "ok")
		})
		m.HandleFunc("/flush", func(w http.ResponseWriter, r *http.Request) { flushAll(); fmt.Fprintln(w, "ok") })
		m.HandleFunc("/anchor", func(w http.ResponseWriter, r *http.Request) { fmt.Fprintln(w, anchorOnce(ctx)) })
		log.Fatal(http.ListenAndServe(":9100", m))
	}()
	sig := make(chan os.Signal, 1)
	signal.Notify(sig, syscall.SIGTERM)
	go func() {
		<-sig
		// Graceful stop: refuse new records so producers rehash, stop the drain, flush what is buffered, report it durable, and write the closing row.
		t0 := time.Now()
		draining.Store(true)
		alsStop()
		for len(walCh) > 0 && time.Since(t0) < 5*time.Second {
			time.Sleep(20 * time.Millisecond)
		}
		time.Sleep(200 * time.Millisecond)
		flushAll()
		for i := 0; i < 600 && buffered.Load() > 0; i++ {
			time.Sleep(100 * time.Millisecond)
			if i%10 == 9 {
				flushAll()
			}
		}
		anchorOnce(ctx)
		_, cerr := pg.Exec(ctx, `update evidence_writers set closed_at=now() where writer=$1`, writerInc)
		close(closeC)
		time.Sleep(300 * time.Millisecond)
		log.Printf("closed buffered=%d in %s row err=%v", buffered.Load(), time.Since(t0).Round(time.Millisecond), cerr)
		os.Exit(0)
	}()
	log.Printf("evidence replica %d writer %s flush %s size %d", replica, writerInc, flushEvery, sizeBound)
	if err := srv.ListenAndServeTLS(os.Getenv("LEAF")+"/tls.crt", os.Getenv("LEAF")+"/tls.key"); err != http.ErrServerClosed {
		log.Fatal(err)
	}
	select {} // the signal handler exits
}

// ---- reads from a named position. A cursor maps a stable position to the delivered ranges. Compaction moves bytes and never positions.
type readRow struct {
	id                         int64
	batch, producer, inc, sess string
	key, digest                string
	seqF, seqL                 int64
	recs                       int
	compacted                  bool
}

func readIndex(ctx context.Context, key string) ([]ev.Part, error) {
	o, err := mc.GetObject(ctx, bucket, key, minio.GetObjectOptions{})
	if err != nil {
		return nil, err
	}
	defer o.Close()
	st, err := o.Stat()
	if err != nil {
		return nil, err
	}
	n := int64(64 << 10)
	if n > st.Size {
		n = st.Size
	}
	for {
		b := make([]byte, n)
		if _, err := o.ReadAt(b, st.Size-n); err != nil && err != io.EOF {
			return nil, err
		}
		p, need, err := ev.ParseIndex(b)
		if err != nil {
			return nil, err
		}
		if need == 0 {
			return p, nil
		}
		n = int64(need)
	}
}

func handleRead(w http.ResponseWriter, req *http.Request) {
	san := sanOf(req)
	if !readers[san] {
		nForbidden.Add(1)
		http.Error(w, "client not allowed", 403)
		return
	}
	q := req.URL.Query()
	tenantID, stream := q.Get("tenant"), q.Get("stream")
	limit, _ := strconv.Atoi(q.Get("limit"))
	if limit <= 0 {
		limit = 10000
	}
	cur := map[string]ev.Ranges{}
	if f := q.Get("from"); f != "" {
		s, _ := url.QueryUnescape(f)
		if err := json.Unmarshal([]byte(s), &cur); err != nil {
			http.Error(w, "bad position", 400)
			return
		}
	}
	if con := q.Get("consumer"); con != "" && q.Get("from") == "" {
		// A registered consumer resumes from its own stored position.
		tx, err := pg.Begin(req.Context())
		if err == nil {
			tx.Exec(req.Context(), `select set_config('app.current_tenant',$1,true)`, tenantID)
			var b []byte
			if tx.QueryRow(req.Context(), `select cursor from evidence_consumers where consumer=$1 and tenant_id=$2 and stream=$3`, con, tenantID, stream).Scan(&b) == nil {
				json.Unmarshal(b, &cur)
			}
			tx.Rollback(req.Context())
		}
	}
	bw := io.Writer(w)
	n, skipped, retried := 0, 0, 0
	type partRead struct {
		key  string
		p    ev.Part
		pkey string
	}
	// A compaction can replace the row under the reader. The reader re-reads the manifest and carries on from its cursor.
attempt:
	for try := 0; try < 5; try++ {
		tx, err := pg.Begin(req.Context())
		if err != nil {
			http.Error(w, err.Error(), 503)
			return
		}
		tx.Exec(req.Context(), `select set_config('app.current_tenant',$1,true)`, tenantID)
		rs, err := tx.Query(req.Context(), `select id,batch_id,producer,incarnation,session,object_key,digest,seq_first,seq_last,records,compacted from manifest where tenant_id=$1 and stream=$2 order by id`, tenantID, stream)
		var rows []readRow
		if err == nil {
			for rs.Next() {
				var r readRow
				if rs.Scan(&r.id, &r.batch, &r.producer, &r.inc, &r.sess, &r.key, &r.digest, &r.seqF, &r.seqL, &r.recs, &r.compacted) == nil {
					rows = append(rows, r)
				}
			}
			rs.Close()
		}
		tx.Rollback(req.Context())
		for _, r := range rows {
			if n >= limit {
				break attempt
			}
			parts := []ev.Part{{Batch: r.batch, Producer: r.producer, Incarnation: r.inc, Session: r.sess, SeqF: r.seqF, SeqL: r.seqL}}
			if r.compacted {
				parts, err = readIndex(req.Context(), r.key)
				if err != nil {
					retried++
					nReadRetry.Add(1)
					time.Sleep(100 * time.Millisecond)
					continue attempt
				}
			}
			for _, p := range parts {
				pk := ev.PosKey(tenantID, stream, p.Producer, p.Incarnation, p.Session)
				if cur[pk].Covers(uint64(p.SeqF), uint64(p.SeqL)) && cur[pk+" gap"] == nil && p.Recs > 0 && false {
					skipped++
					continue
				}
				var o minio.GetObjectOptions
				if p.Len > 0 {
					o.SetRange(p.Off, p.Off+p.Len-1)
				}
				obj, err := mc.GetObject(req.Context(), bucket, r.key, o)
				var b []byte
				if err == nil {
					b, err = io.ReadAll(obj)
					obj.Close()
				}
				if err != nil {
					retried++
					nReadRetry.Add(1)
					time.Sleep(100 * time.Millisecond)
					continue attempt
				}
				g, err := gzip.NewReader(bytes.NewReader(b))
				if err != nil {
					http.Error(w, "bad object", 500)
					return
				}
				z, _ := io.ReadAll(g)
				type it struct {
					a, b uint64
					l    []byte
				}
				var items []it
				for _, l := range bytes.Split(z, []byte{'\n'}) {
					if len(l) == 0 {
						continue
					}
					var x struct{ Sequence, SequenceEnd uint64 }
					json.Unmarshal(l, &x)
					e := x.Sequence
					if x.SequenceEnd > e {
						e = x.SequenceEnd
					}
					items = append(items, it{x.Sequence, e, l})
				}
				sort.Slice(items, func(i, j int) bool { return items[i].a < items[j].a })
				for _, x := range items {
					// a gap record says a range may be lost, and a record for that range can still arrive, so gap records have a cursor of their own
					ck := pk
					if bytes.Contains(x.l[:min(len(x.l), 400)], []byte(`"kind":"gap"`)) {
						ck = pk + " gap"
					}
					if cur[ck].Covers(x.a, x.b) {
						continue
					}
					bw.Write(x.l)
					bw.Write([]byte{'\n'})
					cur[ck] = cur[ck].Add(x.a, x.b)
					n++
				}
			}
		}
		break
	}
	nReadRecords.Add(int64(n))
	trailer, _ := json.Marshal(map[string]any{"kind": "cursor", "cursor": cur, "delivered": n, "partsSkipped": skipped, "retries": retried, "full": n >= limit})
	bw.Write(trailer)
	bw.Write([]byte{'\n'})
}

func handlePosition(w http.ResponseWriter, req *http.Request) {
	if !readers[sanOf(req)] {
		http.Error(w, "client not allowed", 403)
		return
	}
	var x struct {
		Consumer, Tenant, Stream string
		Cursor                   json.RawMessage
	}
	if err := json.NewDecoder(req.Body).Decode(&x); err != nil {
		http.Error(w, "bad", 400)
		return
	}
	tx, err := pg.Begin(req.Context())
	if err == nil {
		defer tx.Rollback(req.Context())
		tx.Exec(req.Context(), `select set_config('app.current_tenant',$1,true)`, x.Tenant)
		_, err = tx.Exec(req.Context(), `insert into evidence_consumers(consumer,tenant_id,stream,cursor) values($1,$2,$3,$4::jsonb) on conflict(consumer,tenant_id,stream) do update set cursor=excluded.cursor, updated_at=now()`, x.Consumer, x.Tenant, x.Stream, string(x.Cursor))
		if err == nil {
			err = tx.Commit(req.Context())
		}
	}
	if err != nil {
		http.Error(w, err.Error(), 503)
		return
	}
	fmt.Fprintln(w, "ok")
}
