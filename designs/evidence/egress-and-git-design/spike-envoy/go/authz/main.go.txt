// Spike ext_authz gRPC server. Decision table keyed on the lease token in
// Proxy-Authorization (Basic user:token):
//
//	token "good"     -> allow, route direct, tenant "acme"
//	token "chain-a"  -> allow, route via upstream proxy A (http_11_proxy cluster)
//	token "chain-b"  -> allow, route via upstream proxy (forwarded CONNECT), with
//	                    the upstream proxy credential injected as Proxy-Authorization
//	token "chain-b-nocred" -> like chain-b but no credential injected
//	token "chain-o"  -> allow, upstream proxy address in x-lenny-upstream (ORIGINAL_DST cluster)
//	token "chain-o2" -> same, pointing at proxy A
//	token "chain-m"  -> upstream proxy address in ext_authz dynamic metadata "upstream" (ORIGINAL_DST metadata_key)
//	token "bad"      -> deny 403
//	no header        -> deny 407 with Proxy-Authenticate
//	host blocked.example.com -> deny 403
//
// Every CheckRequest is logged as one JSON line.
package main

import (
	"context"
	"encoding/base64"
	"encoding/json"
	"log"
	"net"
	"os"
	"strings"
	"sync/atomic"

	corev3 "github.com/envoyproxy/go-control-plane/envoy/config/core/v3"
	authv3 "github.com/envoyproxy/go-control-plane/envoy/service/auth/v3"
	typev3 "github.com/envoyproxy/go-control-plane/envoy/type/v3"
	"google.golang.org/genproto/googleapis/rpc/status"
	"google.golang.org/grpc"
	"google.golang.org/grpc/codes"
	"google.golang.org/protobuf/types/known/structpb"
)

type server struct {
	authv3.UnimplementedAuthorizationServer
	n atomic.Int64
}

func hdr(k, v string) *corev3.HeaderValueOption {
	return &corev3.HeaderValueOption{
		Header:       &corev3.HeaderValue{Key: k, Value: v},
		AppendAction: corev3.HeaderValueOption_OVERWRITE_IF_EXISTS_OR_ADD,
	}
}

func deny(code typev3.StatusCode, reason string, extra ...*corev3.HeaderValueOption) *authv3.CheckResponse {
	md, _ := structpb.NewStruct(map[string]any{"decision": "deny", "reason": reason})
	return &authv3.CheckResponse{
		Status:          &status.Status{Code: int32(codes.PermissionDenied)},
		DynamicMetadata: md,
		HttpResponse: &authv3.CheckResponse_DeniedResponse{DeniedResponse: &authv3.DeniedHttpResponse{
			Status:  &typev3.HttpStatus{Code: code},
			Headers: append([]*corev3.HeaderValueOption{hdr("x-lenny-deny-reason", reason)}, extra...),
			Body:    "denied: " + reason + "\n",
		}},
	}
}

func allow(route, tenant, rule string, extra ...*corev3.HeaderValueOption) *authv3.CheckResponse {
	return allowMD(route, tenant, rule, "", extra...)
}

func allowMD(route, tenant, rule, upstream string, extra ...*corev3.HeaderValueOption) *authv3.CheckResponse {
	m := map[string]any{
		"decision": "allow", "tenant": tenant, "pool": "pool-1", "session": "sess-123", "rule": rule, "route": route,
	}
	if upstream != "" {
		m["upstream"] = upstream
	}
	md, _ := structpb.NewStruct(m)
	ok := &authv3.OkHttpResponse{
		Headers: append([]*corev3.HeaderValueOption{hdr("x-lenny-route", route)}, extra...),
	}
	if !hasHeader(extra, "proxy-authorization") {
		// No replacement credential: strip the client's lease credential.
		ok.HeadersToRemove = []string{"proxy-authorization"}
	}
	return &authv3.CheckResponse{
		Status:          &status.Status{Code: int32(codes.OK)},
		DynamicMetadata: md,
		HttpResponse:    &authv3.CheckResponse_OkResponse{OkResponse: ok},
	}
}

func hasHeader(hs []*corev3.HeaderValueOption, k string) bool {
	for _, h := range hs {
		if h.GetHeader().GetKey() == k {
			return true
		}
	}
	return false
}

func (s *server) Check(ctx context.Context, req *authv3.CheckRequest) (*authv3.CheckResponse, error) {
	n := s.n.Add(1)
	a := req.GetAttributes()
	h := a.GetRequest().GetHttp()
	rec := map[string]any{
		"n":       n,
		"source":  a.GetSource().GetAddress().GetSocketAddress().GetAddress(),
		"srcport": a.GetSource().GetAddress().GetSocketAddress().GetPortValue(),
		"dest":    a.GetDestination().GetAddress().GetSocketAddress().GetAddress(),
		"method":  h.GetMethod(),
		"host":    h.GetHost(),
		"path":    h.GetPath(),
		"scheme":  h.GetScheme(),
		"protocol": h.GetProtocol(),
		"headers": h.GetHeaders(),
		"context": a.GetContextExtensions(),
	}
	b, _ := json.Marshal(rec)
	log.Printf("CHECK %s", b)

	host := h.GetHost()
	if strings.HasPrefix(host, "blocked.example.com") {
		return deny(typev3.StatusCode_Forbidden, "host-not-allowed"), nil
	}
	pa := h.GetHeaders()["proxy-authorization"]
	if pa == "" {
		return deny(typev3.StatusCode_ProxyAuthenticationRequired, "no-credential",
			hdr("proxy-authenticate", `Basic realm="lenny-egress"`)), nil
	}
	raw, err := base64.StdEncoding.DecodeString(strings.TrimPrefix(pa, "Basic "))
	if err != nil {
		return deny(typev3.StatusCode_ProxyAuthenticationRequired, "bad-credential",
			hdr("proxy-authenticate", `Basic realm="lenny-egress"`)), nil
	}
	_, token, _ := strings.Cut(string(raw), ":")
	switch token {
	case "good":
		return allow("direct", "acme", "allow-public"), nil
	case "chain-a":
		return allow("upstream-a", "globex", "via-proxy-a"), nil
	case "chain-b":
		cred := base64.StdEncoding.EncodeToString([]byte("tenant-globex:upstream-secret"))
		return allow("upstream-b", "globex", "via-proxy-b", hdr("proxy-authorization", "Basic "+cred)), nil
	case "chain-b-nocred":
		return allow("upstream-b", "globex", "via-proxy-b-nocred", hdr("x-lenny-dummy", "1")), nil
	case "chain-o":
		// Upstream proxy chosen per request by address: ORIGINAL_DST cluster reads x-lenny-upstream.
		cred := base64.StdEncoding.EncodeToString([]byte("tenant-globex:upstream-secret"))
		return allow("upstream-dyn", "globex", "via-dynamic-proxy",
			hdr("proxy-authorization", "Basic "+cred), hdr("x-lenny-upstream", "198.18.0.13:3128")), nil
	case "chain-m":
		// Upstream proxy address passed only as ext_authz dynamic metadata (no header).
		cred := base64.StdEncoding.EncodeToString([]byte("tenant-globex:upstream-secret"))
		return allowMD("upstream-md", "globex", "via-metadata-proxy", "198.18.0.13:3128",
			hdr("proxy-authorization", "Basic "+cred)), nil
	case "chain-o2":
		// Same, to the no-auth proxy A; credential stripped.
		return allow("upstream-dyn", "initech", "via-dynamic-proxy-a",
			hdr("x-lenny-upstream", "198.18.0.12:3128")), nil
	default:
		return deny(typev3.StatusCode_Forbidden, "lease-invalid"), nil
	}
}

func main() {
	addr := ":9000"
	if v := os.Getenv("ADDR"); v != "" {
		addr = v
	}
	l, err := net.Listen("tcp", addr)
	if err != nil {
		log.Fatal(err)
	}
	g := grpc.NewServer()
	authv3.RegisterAuthorizationServer(g, &server{})
	log.Printf("ext_authz listening on %s", addr)
	log.Fatal(g.Serve(l))
}
